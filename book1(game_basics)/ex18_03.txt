// 3단계: 주사위 두 개를 던져 같은 숫자가 나오면 게임 종료

let dice = ['⚀', '⚁', '⚂', '⚃', '⚄', '⚅'];		// 주사위 유니코드 문자 배열
let num = 0, index1 = 0, index2 = 1;

function setup() {
  createCanvas(400, 200);
}

function draw() {
  background(0);
  
  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);				// 문자열 가운데 정렬
  text('Num: ' + num, width/2, 30);			// 주사위 던진 횟수 표시
  
  textSize(120);
  text(dice[index1], width/2 - 100, height/2);		// 첫 번째 주사위 표시
  text(dice[index2], width/2 + 100, height/2);		// 두 번째 주사위 표시
  
  if (index1 === index2) {				// 두 주사위값이 같을 경우
    fill(255);
    textSize(30);
    text('GAME OVER', width/2, height-40);
    noLoop();						// draw() 함수의 반복 실행 일시 정지
  }
}

function mousePressed() {		// 이벤트함수: 마우스를 누르면 한 번 실행
  num++;				// 주사위 던진 횟수 증가
  index1 = int(random(dice.length));	// 주사위 배열 인덱스(0∼5) 랜덤 선택
  index2 = int(random(dice.length));	// 주사위 배열 인덱스(0∼5) 랜덤 선택
}
