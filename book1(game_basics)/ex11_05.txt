// 완성: 제한시간 내 마우스로 랜덤 이동하는 목표물 5개를 맞추면 다음 단계로 이동, 목표물 크기 감소, 움직임 빨라짐, 제한시간을 초과하면 게임 종료, 스페이스 키를 누르면 재시작

let clapSound, catchSound, cosmosImg, targetImg, timer;
let targetX, targetY, targetSize, rate, tx, ty, score, previousStage, stage = 1;

function preload() {			// setup() 직전에 호출, 외부 파일의 로딩
  clapSound = loadSound('data/claps.mp3');
  catchSound = loadSound('data/catch.mp3');
  cosmosImg = loadImage('data/cosmos.jpg');
  targetImg = loadImage('data/square.png');
}

function setup() {
  createCanvas(600, 600);
  reset();				// 실행 초기: 게임 초기화
}

function draw() {
  image(cosmosImg, 0, 0, width, height);	// 배경 이미지 표시
  
  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);		// 문자열 가운데 정렬
  text('Stage: ' + stage + ', Score: ' + score, width/2, 60);	// 단계, 점수 표시
  
  timerProcess();			// 막대형 타이머 표시 및 게임 종료 처리
  targetProcess();			// 목표물의 표시 및 랜덤 이동
  gamePause();				// 5점 획득 시 단계 이동
}

function timerProcess() {		// 막대형 타이머 표시 및 게임 종료 처리
  fill(255, 0, 0);
  rect(0, 0, timer, 20);			// 막대형 타이머 표시
  
  timer--;				// 타이머 감소
  if (timer <= 0) {			// 타이머 종료된 경우
    fill(255);
    text('GAME OVER', width/2, height/2);

    stage = 1;				// 1단계
    noLoop();				// draw() 함수의 반복 실행 일시 정지
  }
}

function targetProcess() {		// 목표물의 표시 및 랜덤 이동
  targetX = noise(tx) * width;		// targetX값 범위: 0∼width
  targetY = noise(ty) * height;		// targetY값 범위: 0∼height
  image(targetImg, targetX, targetY, targetSize, targetSize);	// 목표물 이미지 표시
  tx += rate;				// 증가율 누적
  ty += rate;				// 증가율 누적
}

function gamePause() {			// 5점 획득 시 단계 이동
  if (score >= 5) {			// 점수가 최소 5점인 경우
    clapSound.play();			// 효과음 재생
    fill(255);
    text('YOU WIN', width/2, height/2 - 30);
    text('Hit the space key to start next stage', width/2, height/2 + 30);

    stage++;				// 단계 증가
    noLoop();				// draw() 함수의 반복 실행 일시 정지
  }
}

function mousePressed() {		// 이벤트함수: 마우스를 누르면 한 번 실행
  if (mouseX > targetX && mouseX < targetX+targetSize && mouseY > targetY && mouseY < targetY+targetSize) {	// 마우스로 목표물 맞춘 경우
    catchSound.play();			// 효과음 재생
    score++;				// 점수 증가
  }
}

function keyPressed() {			// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {			// 스페이스 키를 누르면 게임 초기화, 재실행
    reset();				// 게임 종료 시: 게임 초기화
    loop();					// draw() 함수의 무한 반복 실행
  }
}

function reset() {				// 게임 초기화
  timer = width;				// 타이머 초기화
  tx = 0;					// 증가율 초기화
  ty = 0.5;					// 증가율 초기화
  score = 0;					// 점수 초기화

  if (stage === 1) {				// 1단계인 경우
    targetSize = 50;				// 목표물 크기 초기화
    rate = 0.005;				// 증가율 누적값 초기화
  } else if (stage > 1 && stage !== previousStage) {		// 2단계 이상인 경우
    targetSize -= 2;				// 목표물 크기 감소
    if (targetSize < 30) targetSize = 30;	// 목표물 크기 최소값: 30
    rate += 0.003;				// 증가율 누적값 상승
    if (rate > 0.02) rate = 0.02;		// 증가율 누적값 최대: 0.02
  }
  previousStage = stage;			// 2단계 이상에서 동일 단계는 한 번만 처리
}
