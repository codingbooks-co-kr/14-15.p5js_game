// 두 사각형의 충돌을 감지하고 큰 사각형의 색상 변경-1
// rectMode(CENTER)를 사용한 경우

let x = 150, y = 150, w = 100, h = 100, ww = 50, hh = 50;

function setup() {
  createCanvas(300, 300);
  rectMode(CENTER);		// 사각형의 정가운데를 기준으로 폭과 높이 설정
}

function draw() {
  background(0, 10);		// 페이딩 효과 (투명도 10 부여)
  
  let xx = mouseX;
  let yy = mouseY;

  if (xx+ww/2 > x-w/2 && xx-ww/2 < x+w/2 && yy+hh/2 > y-h/2 && yy-hh/2 < y+h/2) {			// 두 사각형이 충돌한 경우
    fill(255, 0, 0);
  } else {			// 두 사각형이 충돌하지 않은 경우
    fill(0, 255, 0);
  }
  rect(x, y, w, h);		// 캔버스 중앙에 고정된 큰 사각형

  noStroke();			// 부드러운 페이딩 효과를 위해 이동체의 외곽선 제거
  fill(255);
  rect(xx, yy, ww, hh);		// 마우스를 따라 움직이는 작은 사각형
}
