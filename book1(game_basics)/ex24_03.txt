// 3단계: (21×21 미로) 2차원 배열의 홀수 번째 요소에 장벽 만들기-3

let w = 30, maze = [];			// ❶ maze: 1차원 빈 배열의 생성

function setup() {
  createCanvas(630, 630);		// 예) 30(=w:미로단위크기) × 21개 = 630
  designMaze();			// 21×21 미로 디자인
}

function draw() {
  background(0);
  for (let y = 0; y < 21; y++) {	// y는 홀수개 (예: 21개)
    for (let x = 0; x < 21; x++) {	// x는 홀수개 (예: 21개)
      if (maze[y][x] === 0) fill(0);	// 경로: 갈 수 있는 길은 검정색 사각형으로 시각화
      else fill(200);			// 장벽: 가로막힌 벽은 회색 사각형으로 시각화
      rect(x*w, y*w, w, w);		// 30(→w:미로단위크기)*21개 = 630(→캔버스크기)
    }
  }
}

function designMaze() {		// 21×21 미로 디자인
  for (let y = 0; y < 21; y++) {	// y는 홀수개 (예: 21개)
    maze[y] = [];			// ❷ maze: 2차원 빈 배열로 확장
    for (let x = 0; x < 21; x++) {	// x는 홀수개 (예: 21개)
      maze[y][x] = 0;			// ❸ maze: 2차원 배열을 ‘0’(=경로)으로 초기화
    }
  }

  for (let y = 1; y < 21; y += 2) {		// y의 홀수 번째 요소(1,3,5,7,...,19)
    for (let x = 1; x < 21; x += 2) {		// x의 홀수 번째 요소(1,3,5,7,...,19)
      maze[y][x] = 1;				// [1,1] [1,3]... [19,19]에 ‘1’(장벽)로 대체
    }
  }
}
