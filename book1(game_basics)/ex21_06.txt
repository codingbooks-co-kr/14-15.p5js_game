// 6단계: 배경 이미지를 아래쪽으로 지속 이동

let y1, y2;

function preload() {
  cosmosImg = loadImage('data/cosmos.jpg');
}

function setup() {
  createCanvas(600, 600);
  y1 = 0;			// 배경 이미지1 y좌표 초기화
  y2 = -height;		// 배경 이미지2 y좌표 초기화
}

function draw() {
  imageMode(CORNER);	// 이미지 정렬 모드: 이미지의 좌상단 모서리 기준 폭/높이 설정
  image(cosmosImg, 0, y1, width, height); 	// 배경 이미지1 표시
  image(cosmosImg, 0, y2, width, height); 	// 배경 이미지2 표시
  y1 += 2;					// 배경 이미지1 아래쪽으로 이동
  y2 += 2;					// 배경 이미지2 아래쪽으로 이동
  if (y1 > height) y1 = -height;		// 이미지1: 캔버스를 벗어나면 y좌표 재설정
  if (y2 > height) y2 = -height;		// 이미지2: 캔버스를 벗어나면 y좌표 재설정
}
