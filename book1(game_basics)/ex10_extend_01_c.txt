// frameCount 없이 시간 측정: 0.5초마다 원형 생성

let timer = 0;			// 전역변수 초기화

function setup() {
  createCanvas(300, 300);
  background(0);
}

function draw() {
  timer += 1/30;		// draw() 초당 60회 실행 → timer: 초당 (1/30)*30회≈0.5초
  //console.log(timer);		// (값 확인용) timer 누적값은 실수값이므로 1이 되지 않음
  if (timer > 1) {		// 0.5초마다 if문은 true가 됨
    timer = 0;			// 누적을 위해 다시 초기화
    fill(random(width), random(width), random(width), 200);
    ellipse(random(width), random(height), 50, 50);
  }
}
