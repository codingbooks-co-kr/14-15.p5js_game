// 두 원의 충돌을 감지하고 큰 원의 색상 변경

let x = 150, y = 150, d1 = 100, d2 = 50;

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0, 10);		// 페이딩 효과 (투명도 10 부여)

  let xx = mouseX;
  let yy = mouseY;

  let distance = dist(x, y, xx, yy);
  if (distance < (d1 + d2) / 2) {	// 두 원이 충돌한 경우
    fill(255, 0, 0);
  } else {				// 두 원이 충돌하지 않은 경우
    fill(0, 255, 0);
  }
  ellipse(x, y, d1, d1);			// 캔버스 중앙에 고정된 큰 원

  noStroke();				// 부드러운 페이딩 효과를 위해 이동체의 외곽선 제거
  fill(255);
  ellipse(xx, yy, d2, d2);		// 마우스를 따라 움직이는 작은 원
}
