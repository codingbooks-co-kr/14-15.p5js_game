// random() 및 constrain() 함수를 사용하여 원의 불규칙한 움직임을 구현

let x = 250, y = 250, speed = 10;

function setup() {
  createCanvas(500, 500);
  background(0);
}

function draw() {
  x += random(-speed, speed);		// (-10, 10) 사이의 랜덤값을 x에 누적
  y += random(-speed, speed);		// (-10, 10) 사이의 랜덤값을 y에 누적
  ellipse(x, y, 30, 30);
  x = constrain(x, 0, width);			// x값의 범위 제한: 0∼width
  y = constrain(y, 0, height);			// y값의 범위 제한: 0∼height
}
