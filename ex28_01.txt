// [본문 코드] 문자열, 원, 사각형, 삼각형 객체의 좌우 반사

let sh0, sh1, sh2, sh3;

function setup() {
  createCanvas(300, 300);
  sh0 = new Shape(255, 0, 30, 30, 1);	// 상위 객체 생성 및 초기화
  sh1 = new Circle(255, 0, 70, 30, 2);	// 하위 객체(원) 생성 및 초기화
  sh2 = new Square(255, 0, 140, 30, 3);	// 하위 객체(사각형) 생성 및 초기화
  sh3 = new Triangle(255, 0, 210, 30, 4);	// 하위 객체(삼각형) 생성 및 초기화
}

function draw() {
  background(0);
  sh0.display();				// 상위 메소드 호출
  sh0.move();				// 상위 메소드 호출
  sh1.display();				// 메소드 오버라이딩
  sh1.move();				// 상속
  sh2.display();				// 메소드 오버라이딩
  sh2.move();				// 상속
  sh3.display();				// 메소드 오버라이딩
  sh3.move();				// 상속
}

// [클래스 코드]

class Shape {				// 상위 클래스(Shape) 
  constructor(c, x, y, d, speed) {	// 상위 생성자: 속성 추가 및 초기화
    this.c = c;				// 속성: 색상 초기화
    this.x = x;				// 속성: x축 위치좌표 초기화
    this.y = y;				// 속성: y축 위치좌표 초기화
    this.d = d;				// 속성: 크기 초기화
    this.speed = speed;		// 속성: 속도 초기화
  }
  display() {				// 상위 메소드: 문자열 출력
    fill(this.c, this.c, 0);
    textSize(this.d);
    text("Shape", this.x, this.y);
  }
  move() {				// 상위 메소드: 좌우 이동 및 반사
    this.x += this.speed;		// x축 속도 누적 → x축 거리 이동 (부호: 방향성)
    if (this.x < 0 || this.x > width) {		// 캔버스의 좌우 경계에서 충돌하는 경우
      this.speed *= -1;			// 속도의 방향 반전, 즉 반사
    }
  }
}

class Circle extends Shape {		// 하위 클래스(Circle): 상위 클래스(Shape) 상속
  constructor(c, x, y, d, speed) {	// 하위 생성자
    super(c, x, y, d, speed);		// 상위 생성자 호출
  }
  display() {				// 메소드 오버라이딩
    fill(this.c, 0, 0);
    ellipse(this.x, this.y, this.d, this.d);
  }
}

class Square extends Shape {		// 하위 클래스(Square): 상위 클래스(Shape) 상속
  constructor(c, x, y, d, speed) {	// 하위 생성자
    super(c, x, y, d, speed);		// 상위 생성자 호출
  }
  display() {				// 메소드 오버라이딩
    fill(0, this.c, 0);
    rect(this.x, this.y, this.d, this.d);
  }
}

class Triangle extends Shape {		// 하위 클래스(Triangle): 상위 클래스(Shape) 상속
  constructor(c, x, y, d, speed) {	// 하위 생성자
    super(c, x, y, d, speed);		// 상위 생성자 호출
  }
  display() {				// 메소드 오버라이딩
    fill(0, 0, this.c);
    triangle(this.x, this.y, this.x-this.d, this.y+this.d, this.x+this.d, this.y+this.d);
  }
}
