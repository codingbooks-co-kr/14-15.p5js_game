// p5.js: 1초마다 랜덤 위치로 변경되는 목표물을 마우스로 맞추면 1, 맞추지 못하면 0 송신
// 10.2절 “매초 랜덤 위치로 변경되는 목표물 맞추기” 참조

let port = new p5.WebSerial();		// 시리얼 통신을 위한 객체 생성
let targetX, targetY, value, targetSize = 50, num = 0, score = 0;
let interval, currentTime, previousTime = 0, setTime = 1000;

function setup() {
  createCanvas(600, 600);
  port.getPorts();			// 포트 사용 가능 여부 확인
  port.on('noport', makePortButton);	// 기존 선택한 포트가 없으면 makePortButton() 호출
  port.on('portavailable', openPort);	// 사용 가능한 포트가 있으면 openPort() 자동 호출
  port.on('data', serialEvent);		// 데이터 수신: ‘data’ 이벤트 발생→serialEvent() 호출

  targetX = random(targetSize, width-targetSize);	// 목표물의 x축 좌표 랜덤 변경
  targetY = random(targetSize, height-targetSize);	// 목표물의 y축 좌표 랜덤 변경
}

function draw() {
  background(0);

  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);			// 문자열 가운데 정렬
  text('Number: ' + num + ', Score: ' + score, width/2, 30);	// 출현횟수, 점수 표시

  fill(255);
  ellipse(targetX, targetY, targetSize, targetSize);	// 목표물 표시

  currentTime = millis();			// 현재시간 설정
  interval = currentTime - previousTime; 	// 시간차(interval) = 현재시간 - 이전시간
  if (interval > setTime) {			// setTime(=1000msec) 마다 실행
    value = 0;					// 송신값: 0
    previousTime = millis();			// 이전시간을 현재시간으로 갱신
    targetX = random(targetSize, width-targetSize);	// 목표물의 x축 좌표 랜덤 변경
    targetY = random(targetSize, height-targetSize);	// 목표물의 x축 좌표 랜덤 변경
    num++;					// 목표물 출현 횟수 증가
  }
}

function mousePressed() {			// 이벤트함수: 마우스를 누르면 자동 실행
  let distance = dist(targetX, targetY, mouseX, mouseY);	// 목표물∼마우스 거리
  if (distance < targetSize/2) {			// 원형 내부에서 마우스를 누른 경우
    value = 1;					// 송신값: 1
    score++;					// 점수 상승
  }
}

function serialEvent() {				// 콜백함수: 수신값이 있으면 자동 호출
  let signal = port.read();			// 수신: 통신시작신호('A'), 통신제어신호('B')
  port.write(value);				// 송신: 0 또는 1
}

function makePortButton() {			// 콜백함수: 포트 첫연결 시 포트선택버튼 생성
  let portButton = createButton('포트 선택');	// 버튼 생성
  portButton.mousePressed(choosePort);	// 버튼 누르면 choosePort() 자동 호출
  function choosePort() {			// 콜백함수
    port.requestPort();				// 시리얼 포트 선택창 오픈
  }
}

function openPort() {				// 콜백함수: 포트 오픈
  port.open().then(initiateSerial);		// 포트를 오픈 후 initiateSerial() 자동 호출
  function initiateSerial() {			// 콜백함수
    console.log('포트 오픈');
  }
}
