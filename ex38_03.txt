// 완성: 방향키로 플레이어를 격자(grid) 단위로 이동하면서 장애물을 피해 최상단 라인에 도착하면 경과 시간 표시, 단계가 올라갈수록 장애물의 이동 속도 증가, 적과 충돌하면 게임 종료, 스페이스 키를 누르면 재시작

let clapSound, burstSound, cosmosImg, burstImg, playerImg, player, obstacles;
let interval, currentTime, endTime, gridSize, speedMax;
let previousStage, stage = 1, firstStep = true;

function preload() {		// setup() 직전에 호출, 외부 파일의 로딩
  clapSound = loadSound('data/claps.mp3');
  burstSound = loadSound('data/burst.mp3');
  cosmosImg = loadImage('data/cosmos.jpg');
  burstImg = loadImage('data/burst.png');
  playerImg = loadImage('data/rocket2.png');
}

function setup() {
  createCanvas(600, 500);
  reset();			// 실행 초기: 게임 초기화
}

function draw() {
  image(cosmosImg, 0, 0, width, height);	// 배경 이미지 표시

  currentTime = millis() / 1000;		// 실행 이후의 경과시간(초) 측정
  interval = currentTime - endTime; 		// 시간차(interval) = 현재시간 - 종료시간

  fill(255);
  text('Stage: ' + stage + ', Driving Time: ' + nf(interval, 3, 1) + ' sec', width/2, 30);

  fill(0, 255, 255, 100);
  rect(0, 0, width, gridSize);			// 도착 라인 (row10)
  rect(0, height - gridSize, width, gridSize);	// 출발 라인 (row1)
  
  player.display();				// 플레이어 객체 표시

  if (player.y < height - gridSize*9) {		// 플레이어가 캔버스의 최상단에 도착한 경우
    clapSound.play();				// 효과음 재생
    fill(255);
    text('YOU WIN', width/2, height/2 - 30);
    text('Hit the space key to start next stage', width/2, height/2 + 30);

    stage++;					// 단계 상승
    noLoop();					// draw() 함수의 반복 실행 일시 정지
  }

  for (let i = 0; i < obstacles.length; i++) {
    obstacles[i].display();			// 장애물 객체 표시
    obstacles[i].move();				// 장애물 객체 이동

    if (player.hit(obstacles[i])) {			// 플레이어∼장애물 충돌한 경우
      player.explode();				// 플레이어 폭발하는 효과 표시
      fill(255);
      text('GAME OVER', width/2, height/2);

      stage = 1;				// 1단계
      noLoop();				// draw() 함수의 반복 실행 일시 정지
    }
  }
}

function keyPressed() {		// 스페이스키: 게임 초기화, 방향키: 격자(grid) 단위 이동
  if (key === ' ') {		// 스페이스키를 누르는 경우
    reset();			// 게임 종료 시: 게임 초기화
    loop();			// draw() 함수의 무한 반복 실행
  }

  if (keyIsDown(LEFT_ARROW)) player.move(-1, 0);		// 한 격자(grid)씩 왼쪽 이동
  if (keyIsDown(RIGHT_ARROW)) player.move(1, 0);		// 한 격자(grid)씩 오른쪽 이동
  if (keyIsDown(UP_ARROW)) player.move(0, -1);		// 한 격자(grid)씩 위쪽 이동
  if (keyIsDown(DOWN_ARROW)) player.move(0, 1);		// 한 격자(grid)씩 아래쪽 이동
}

function reset() {			// 게임 초기화
  if (stage === 1) {			// 1단계인 경우
    speedMax = 2;			// 장애물의 이동 최대속도 초기화
  } else if (stage > 1 && stage !== previousStage) {		// 2단계 이상인 경우
    speedMax += 0.5;			// 장애물의 이동 최대속도 증가
  }
  previousStage = stage;		// 2단계 이상에서 동일 단계는 한 번만 처리

  gridSize = 50;			// 격자 크기 초기화
  player = new Player(gridSize);	// 플레이어 객체 생성 및 초기화

  let index = 0, rows = 2, turn = 1;		// 지역변수
  obstacles = [];				// 장애물: 빈 객체 배열로 초기화
  for (let i = 0; i < 8; i++) {    		// 라인(row2∼row9), 출발:row1, 도착:row10
    let speedUp = random(0.5, speedMax) * turn;	// 라인별 장애물 속도, 방향(turn)
    for (let j = 0; j < 3; j++) {			// 라인별 장애물 객체 수는 3개
      let x = j * 180 + random(50, 100);	// 장애물 객체 x축 좌표
      let y = height - (gridSize * rows);	// 장애물 객체 y축 좌표
      obstacles[index] = new Obstacle(x, y, gridSize, speedUp);	// 객체생성,초기화
      index++;
    }
    rows++;					// 장애물 객체의 생성 라인 증가
    turn *= -1;				// 장애물 객체의 이동 방향 반전
  }

  if (firstStep) endTime = 0;			// 첫 시작만 endTime은 0으로 초기화
  else endTime = millis() / 1000;		// 두 번째부터 현재시간(초)으로 초기화
  firstStep = false;

  textSize(30);
  textAlign(CENTER, CENTER);			// 문자열 가운데 정렬
}

class Obstacle {				// 장애물 클래스	
  constructor(x, y, gridSize, speedUp) {	// 생성자: 속성 추가 및 초기화
    this.x = x;					// 장애물 x축 좌표 초기화
    this.y = y;					// 장애물 y축 좌표 초기화
    this.size = gridSize;			// 장애물 크기 초기화
    this.speed = speedUp;			// 장애물 속도 초기화
  }
  display() {					// 메소드: 장애물 객체 표시
    fill(255, 100);
    rect(this.x, this.y, this.size, this.size);
  }
  move() {					// 메소드: 장애물 객체 이동
    this.x += this.speed;			// x축속도누적 → x축거리이동 (부호:방향성)
    if (this.speed > 0 && this.x > width) {	// 장애물이 캔버스 오른쪽 끝에 이르면
      this.x = -this.size;			// 캔버스 왼쪽에서 이동 시작
    } else if (this.speed < 0 && this.x < -this.size) {	// 장애물이 캔버스 왼쪽 끝에 이르면
      this.x = width;				// 캔버스 오른쪽에서 이동 시작
    }
  }
}

class Player {				// 플레이어 클래스 
  constructor(gridSize) {		// 생성자: 속성 추가 및 초기화
    this.x = width / 2;		// 플레이어 x좌표 초기화
    this.y = height - gridSize;		// 플레이어 y좌표 초기화
    this.size = gridSize;		// 플레이어 크기 초기화
    this.speed = gridSize;		// 플레이어 속도 초기화
  }
  display() {				// 메소드: 플레이어 객체 표시
    image(playerImg, this.x, this.y, this.size, this.size);	// 플레이어 이미지 표시
  }
  explode() {				// 메소드: 플레이어 객체 폭발 효과 표시
    burstSound.play();			// 효과음 재생
    image(burstImg, this.x, this.y, this.size*2, this.size*2);	// 폭발 이미지 표시
  }
  move(turnX, turnY) {			// 메소드: 플레이어 객체 이동
    this.x += this.speed * turnX;	// x축 속도 누적 → x축 거리 이동 (turnX: 방향성)
    this.y += this.speed * turnY;	// y축 속도 누적 → y축 거리 이동 (turnY: 방향성)
    this.x = constrain(this.x, 0, width-this.size);	// x축 이동 제한
    this.y = constrain(this.y, 0, height-this.size);	// y축 이동 제한
  }
  hit(other) {				// 사각형 장애물과 충돌 여부 판단(→5.4절 참조)
    if (this.x < other.x + other.size && this.x + this.size > other.x && this.y < other.y + other.size && this.y + this.size > other.y) {	// 두 객체의 충돌 시
      return true;
    } else {							// 두 객체의 미충돌 시
      return false;
    }
  }
}
