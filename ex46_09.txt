// 조도센서값 읽고 출력

#define LIGHT A3

void setup() {
  pinMode(LIGHT, INPUT);		// 조도센서핀을 입력으로 사용
  Serial.begin(9600);			// 전송속도: 9,600bps
}

void loop() {
  int value = analogRead(LIGHT);	// 조도센서값 읽기
  Serial.print("Light: ");			// 시리얼 모니터상에 출력
  Serial.println(value);			// 시리얼 모니터상에 조도센서값 출력
  delay(100);
}
