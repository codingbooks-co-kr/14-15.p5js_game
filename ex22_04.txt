// 완성: 키보드 방향키로 플레이어를 상하좌우 이동, 스페이스 키를 눌러 총알 발사, 낙하하는 목표물을 맞추면 점수 상승, 목표물이 캔버스 화면을 벗어나면 게임 종료, ‘r’키를 눌러 재시작

let clapSound, burstSound, burstImg, cosmosImg, targetImg, playerImg, score;
let y1, y2, targets, bullets, player, timer, previousStage, stage = 1;

function preload() {		// setup() 직전에 호출, 외부 파일의 로딩
  clapSound = loadSound('data/claps.mp3');
  burstSound = loadSound('data/burst.mp3');
  burstImg = loadImage('data/burst.png');
  cosmosImg = loadImage('data/cosmos.jpg');
  targetImg = loadImage('data/circle1.png');
  playerImg = loadImage('data/rocket2.png');
}

function setup() {
  createCanvas(500, 600);
  reset();			// 실행 초기: 게임 초기화
}

function draw() {
  imageMode(CORNER);	// 이미지 정렬 모드: 이미지의 좌상단 모서리 기준 폭/높이 설정
  image(cosmosImg, 0, y1, width, height); 	// 배경 이미지1 표시
  image(cosmosImg, 0, y2, width, height); 	// 배경 이미지2 표시
  y1 += 2;					// 배경 이미지1 아래로 이동
  y2 += 2;					// 배경 이미지2 아래로 이동
  if (y1 > height) y1 = -height;		// 이미지1: 캔버스를 벗어나면 y좌표 재설정
  if (y2 > height) y2 = -height;		// 이미지2: 캔버스를 벗어나면 y좌표 재설정
  
  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);			// 문자열 가운데 정렬
  text('Stage: ' + stage + ', Score: ' + score, width/2, 30);	// 단계, 점수 표시

  imageMode(CENTER);		// 이미지정렬모드: 이미지의 가운데에서 폭/높이 설정
  playerControl();			// 플레이어 상하좌우 이동 제어 
  targetProcess();			// 목표물 객체의 생성, 추가, 표시, 이동, 게임 종료
  shooting();				// 총알 발사 후 목표물 맞추면 점수 상승, 단계 이동
}

function playerControl() {		// 플레이어 상하좌우 이동 제어 
  image(playerImg, player.x, player.y, player.size, player.size);	  // 플레이어 이미지 표시

  if (keyIsDown(LEFT_ARROW)) {	// 왼쪽 방향키를 누른 경우
    player.x -= player.speed;		// x축 속도(-부호: 왼쪽) 누적 → 왼쪽 이동 
  }
  if (keyIsDown(RIGHT_ARROW)) {	// 오른쪽 방향키를 누른 경우
    player.x += player.speed;		// x축 속도(+부호: 오른쪽) 누적 → 오른쪽 이동 
  }
  if (keyIsDown(UP_ARROW)) {	// 위쪽 방향키를 누른 경우
    player.y -= player.speed;		// y축 속도(-부호: 위쪽) 누적 → 위쪽 이동 
  }
  if (keyIsDown(DOWN_ARROW)) {	// 아래쪽 방향키를 누른 경우
    player.y += player.speed;		// y축 속도(+부호: 아래쪽) 누적 → 아래쪽 이동 
  }
  player.x = constrain(player.x, 0, width);	// 좌우 이동 제한
  player.y = constrain(player.y, 0, height);	// 상하 이동 제한
}

function targetProcess() {		// 목표물 객체의 생성, 추가, 표시, 이동, 게임 종료
  if (frameCount % timer === 0) {	// timer 주기로 목표물 객체 생성
    let target = { x: random(width), y: 0, size: 30, speed: 2 };
    targets.push(target);
  }

  for (let i = targets.length-1; i >= 0; i--) {		// 목표물 객체별 역순 반복 처리
    image(targetImg, targets[i].x, targets[i].y, targets[i].size, targets[i].size);	// 목표물

    targets[i].y += targets[i].speed;	// 목표물의 y축 속도 누적 → 아래쪽 이동
    if (targets[i].y > height) {		// 목표물이 캔버스 화면을 벗어나는 경우
      fill(255);
      text('Game Over', width/2, height/2);

      stage = 1;			// 1단계
      noLoop();				// draw() 함수의 반복 실행 일시 정지
    }
  }
}

function shooting() {			// 총알 발사 후 목표물 맞추면 점수 상승, 단계 이동
  for (let i = bullets.length-1; i >= 0; i--) {		// 총알 객체별 역순 반복 처리
    fill(255, 0, 0);
    ellipse(bullets[i].x, bullets[i].y, bullets[i].size, bullets[i].size);

    bullets[i].y -= bullets[i].speed;	// 총알 객체는 위쪽 방향으로 이동(=발사)
    if (bullets[i].y < 0) {		// 총알 객체가 캔버스 화면을 벗어난 경우
      bullets.splice(i, 1);		// bullets[i] 총알 객체 제거
    } else {				// 총알이 발사되어 캔버스 안에서 이동 중인 경우
      for (let j = targets.length-1; j >= 0; j--) {	// 목표물 객체별 역순 반복 처리
        let distance = dist(bullets[i].x, bullets[i].y, targets[j].x, targets[j].y);
        if (distance < (bullets[i].size + targets[j].size) / 2) {	 // 총알∼목표물 충돌 검사
          burstSound.play();		// 효과음 재생
          image(burstImg, targets[j].x, targets[j].y, targets[j].size*2, targets[j].size*2);
          targets.splice(j, 1);		// targets[j] 목표물 객체 제거
          bullets.splice(i, 1);		// bullets[i] 총알 객체 제거
          score++;			// 점수 증가
          break;			// 반복문 빠져나옴
        }
      }
    }
  }

  if (score >= 7) {			// 목표물 최소 7개 맞추면 단계 이동
    clapSound.play();			// 효과음 재생
    noStroke();
    fill(255);
    text('YOU WIN', width/2, height/2 - 30);
    text('Hit the r key to start next stage', width/2, height/2 + 30);

    stage++;				// 단계 증가
    noLoop();				// draw() 함수의 반복 실행 일시 정지
  }
}

function keyPressed() {			// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {			// 스페이스 키를 누르면 총알 발사
    let bullet = { x: player.x, y: player.y, size: 10, speed: 10 };	// 객체생성,초기화
    bullets.push(bullet);			// 생성된 bullet 객체를 bullets 배열에 추가
  }
  if (key === 'r') {			// 'r' 키를 누르면 게임 초기화
    reset();				// 게임 종료 시: 게임 초기화
    loop();				// draw() 함수의 무한 반복 실행
  }
}

function reset() {			// 게임 초기화
  y1 = 0;				// 배경 이미지1 y좌표 초기화
  y2 = -height;			// 배경 이미지2 y좌표 초기화
  targets = [];				// 목표물 객체: 빈 배열로 초기화
  bullets = [];				// 총알 객체: 빈 배열로 초기화
  player = { x: width/2, y: height/2, size: 30, speed: 3 };	  // 플레이어 객체 초기화
  score = 0;				// 점수 초기화

  if (stage === 1) {			// 1단계인 경우
    timer = 60;			// 목표물 생성 타이머 초기화
  } else if (stage > 1 && stage !== previousStage) {		// 2단계 이상인 경우
    timer -= 5;			// 목표물 생성 타이머 감소 → 목표물이 더 빨리 생성
    if (timer < 20) timer = 20;		// 목표물 생성 타이머 최소값: 20
  }
  previousStage = stage;		// 2단계 이상에서 동일 단계는 한 번만 처리
}
