// 아두이노: 마우스의 좌우 이동값(→정수: 0∼255)을 수신 후 LED 밝기 제어

#define LED 6				// LED 핀번호는 6

void setup() {
  pinMode(LED, OUTPUT);		// LED핀을 출력으로 사용
  Serial.begin(9600);			// 전송속도: 9,600bps
  while (Serial.available() == 0) {	// 수신버퍼에 수신값이 없는 동안 반복 실행
    Serial.write('A');			// 시작신호('A') 전송
    delay(300);     			// 0.3초 대기
  }
}

void loop() { }

void serialEvent() {    			// 이벤트함수: 수신값이 있으면 자동 호출
  int value = Serial.read(); 		// 수신: 마우스의 좌우 이동값(0∼255)
  analogWrite(LED, value);		// 수신값으로 LED 밝기 제어
  Serial.write('B');        		// 송신: 통신제어신호('B')
}
