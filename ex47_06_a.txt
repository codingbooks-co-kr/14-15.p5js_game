// 아두이노: 목표물 획득 여부값 수신 후 부저음 출력 제어, 가변저항값 송신

#define VARIABLE_R A1		// 가변저항 핀번호: A1
#define BUZZER 13			// 부저 핀번호: 13

void setup() {
  pinMode(VARIABLE_R, INPUT);	// 가변저항핀을 입력으로 사용
  pinMode(BUZZER, OUTPUT);		// 부저핀을 출력으로 사용
  Serial.begin(9600);			// 전송속도: 9,600bps
  while (Serial.available() == 0) {	// 수신버퍼에 수신값이 없는 동안 반복 실행
    Serial.write('A');			// 시작신호('A') 전송
    delay(300);     			// 0.3초 대기
  }
}

void loop() { }

void serialEvent() {    				// 이벤트함수: 수신값이 있으면 자동 호출
  int catchTarget = Serial.read(); 		// 수신: 목표물획득여부(0 또는 1)
  if (catchTarget == 0) noTone(BUZZER);	// 목표물 미 획득 시 무음
  else tone(BUZZER, 440);			// 목표물 획득 시 440Hz 주파수음 출력

  int value = analogRead(VARIABLE_R);	// 가변저항값 읽기
  value = map(value, 0, 1023, 0, 255);	// 가변저항값 범위를 0∼255로 변환
  Serial.write(value);				// 송신: 가변저항값(0∼255)
}
