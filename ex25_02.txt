// [본문 코드] 원 좌우 반사-3

let c;           	// 객체 선언

function setup() {
  createCanvas(300, 200);
  c = new Circle();	// 객체 생성
}

function draw() {
  background(0, 10);	// 페이딩 효과
  c.display();		// 메소드 호출
  c.move();		// 메소드 호출
}

// [클래스 코드]

class Circle { 			// 클래스
  constructor() {     		// 생성자
    this.c = color(255);
    this.x = width/2;
    this.y = height/2;
    this.d = 30;
    this.speed = 1;
  }
  display() {			// 메소드
    noStroke();
    fill(this.c);
    ellipse(this.x, this.y, this.d, this.d);
  }
  move() {			// 메소드
    this.x += this.speed;
    if (this.x < 0 || this.x > width) {
      this.speed *= -1;
    }
  }
}
