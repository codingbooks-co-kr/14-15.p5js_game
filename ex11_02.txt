// 2단계: 랜덤하게 움직이는 사각형 목표물을 마우스로 맞추면 색상 변경

let col, targetX, targetY, targetSize = 50, tx = 0, ty = 0.5;

function setup() {
  createCanvas(500, 500);
  col = color(random(255), random(255), random(255));	// RGB 랜덤 색상값 설정
}

function draw() {
  background(0, 10);				// 페이딩 효과 (투명도 10 부여)
  
  targetX = noise(tx) * width;			// targetX 랜덤값 범위: 0∼width
  targetY = noise(ty) * height;			// targetY 랜덤값 범위: 0∼height

  stroke(255);
  fill(col);					// 색상값(col) 표시
  rect(targetX, targetY, targetSize, targetSize);	// 랜덤 이동하는 사각형 목표물 표시
  
  tx += 0.01;					// 증가율 누적
  ty += 0.01;					// 증가율 누적
}

function mousePressed() {			// 이벤트함수: 마우스를 누르면 한 번 실행
  if (mouseX > targetX && mouseX < targetX+targetSize && mouseY > targetY && mouseY < targetY+targetSize) {		// 마우스로 목표물 맞춘 경우
    col = color(random(255), random(255), random(255));	// RGB 랜덤 색상값 설정
  }
}
