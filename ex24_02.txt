// 2단계: (5×5 미로) 2차원 배열의 홀수 번째 요소에 장벽 만들기-2

let w = 50, maze = [];			// ❶ maze: 1차원 빈 배열의 생성

function setup() {
  createCanvas(250, 250);		// 50(=w:미로단위크기) × 5개 = 250
  for (let y = 0; y < 5; y++) {		// y는 홀수개 (예: 5개)
    maze[y] = [];			// ❷ maze: 2차원 빈 배열로 확장
    for (let x = 0; x < 5; x++) {	// x는 홀수개 (예: 5개)
      maze[y][x] = 0;			// ❸ maze: 2차원 배열을 ‘0’(경로)으로 초기화
    }
  }

  for (let y = 1; y < 5; y += 2) {	// y = 1, 3 (홀수 번째 요소)
    for (let x = 1; x < 5; x += 2) {	// x = 1, 3 (홀수 번째 요소}
      maze[y][x] = 1;			// [1,1] [1,3] [3,1] [3,3] 요소에 ‘1’(장벽)로 초기화
    }
  }
}

function draw() {
  background(0);
  for(let y = 0 ; y < 5; y++) {		// y는 홀수개 (예: 5개)
    for(let x = 0 ; x < 5; x++) {	// x는 홀수개 (예: 5개)
      if (maze[y][x] === 0) fill(0);	// 경로: 갈 수 있는 길은 검정색 사각형으로 시각화
      else fill(200);			// 장벽: 가로막힌 벽은 회색 사각형으로 시각화
      rect(x*w, y*w, w, w);		// 50(→w:미로단위크기) * 5개 = 250(→캔버스크기)
    }
  }
}
