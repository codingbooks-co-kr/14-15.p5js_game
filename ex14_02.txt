// 2단계: 스페이스 키를 눌러 플레이어를 한 번 제자리 점프

let playerX, playerY, playerSize = 50, playerSpeed = 0;
let gravity = 0.6, jumpPower = -10, jump = false;

function setup() {
  createCanvas(300, 200);
  playerX = width / 2;				// 플레이어 x축 좌표 초기화
  playerY = height - playerSize/2;		// 플레이어 y축 좌표 초기화
}

function draw() {
  background(0, 10);			// 페이딩 효과 (투명도 10 부여)
  
  noStroke();				// 부드러운 페이딩 효과를 위해 이동체의 외곽선 제거
  fill(0, 255, 0);
  ellipse(playerX, playerY, playerSize, playerSize);	// 플레이어 표시
  
  playerSpeed += gravity;		// 속도에 중력 누적 → 낙하 시 가속 
  playerY += playerSpeed;		// y축 속도(부호: 방향성) 누적 → y축 거리 이동 

  if (playerY > height - playerSize/2) {  	// 플레이어가 캔버스 바닥에 닿은 경우
    playerY = height - playerSize/2;		// 플레이어를 캔버스 바닥에 위치시킴
    playerSpeed = 0;				// 속도는 지속 증가하므로 0으로 초기화
    jump = false;			// ❶플레이어가 캔버스 바닥에 있을 때 점프 가능
  }
}

function keyPressed() {			// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ' && !jump) {		// 스페이스키를 누르고 플레이어가 바닥에 있을 때
    playerSpeed = jumpPower;		// 점프 속도는 –10, 즉 윗방향으로 점프
    jump = true;			// ❷플레이어가 점프한 상태에서는 재점프 불가능
  }
}
