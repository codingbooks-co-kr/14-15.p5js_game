// 완성: 낙하하는 목표물 잡기, 놓치면 게임 종료, 스페이스키를 누르면 재시작

let catchSound, burstSound, burstImg, cosmosImg, targetImg, playerImg, score;
let playerX, playerY, w, h, targetX, targetY, targetSize, targetSpeed;

function preload() {		// setup() 직전에 호출, 외부 파일의 로딩
  catchSound = loadSound('data/catch.mp3');
  burstSound = loadSound('data/burst.mp3');
  burstImg = loadImage('data/burst.png');
  cosmosImg = loadImage('data/cosmos.jpg');
  targetImg = loadImage('data/circle1.png');
  playerImg = loadImage('data/brick.png');
}

function setup() {
  createCanvas(600, 400);
  reset();			// 실행 초기: 게임 초기화 함수 호출
}

function draw() {
  image(cosmosImg, width/2, height/2, width, height);	// 배경 이미지 표시
  
  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);			// 문자열 가운데 정렬
  text('Score: ' + score, width/2, 30);		// 점수 표시
  
  display();					// 목표물 및 플레이어 표시 
  move();					// 목표물의 이동 및 게임 종료 처리
  catchTarget();					// 목표물을 잡았을 때 처리
}

function display() {				// 목표물 및 플레이어 표시 
  image(targetImg, targetX, targetY, targetSize, targetSize);	// 목표물 이미지 표시

  playerX = constrain(mouseX, w/2, width-w/2);		// 플레이어 x축 이동 제한
  playerY = height - h;					// 플레이어 y축 위치 고정
  image(playerImg, playerX, playerY, w, h);			// 플레이어 이미지 표시
}

function move() {		// 목표물의 이동 및 게임 종료 처리
  targetY += targetSpeed;	// y축 속도 누적 → 낙하
  if (targetY > height) {	// 목표물이 캔버스 바닥을 벗어난 경우
    burstSound.play();		// 효과음 재생
    image(burstImg, targetX, targetY, targetSize*2, targetSize*2);	// 폭발 이미지 표시

    fill(255);
    text('GAME OVER', width/2, height/2 - 30);
    text('Hit the space key to restart', width/2, height/2 + 30);
    noLoop();			// draw() 함수의 반복 실행 일시 정지
  }
}

function catchTarget() {		// 목표물을 잡았을 때 처리
  if (targetX > playerX-w/2 && targetX < playerX+w/2 && targetY > playerY) {
    catchSound.play();		// 효과음 재생
    targetX = random(targetSize/2, width-targetSize/2);	// 목표물 x축 좌표 초기화
    targetY = 0;						// 목표물 y축 좌표 초기화
    targetSpeed++;		// 낙하 속도 증가
    score++;			// 점수 증가
  }
}

function keyPressed() {		// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {		// 스페이스 키를 누르면 게임 초기화, 재실행
    reset();			// 게임 종료 시: 게임 초기화 함수 호출
    loop();			// draw() 함수의 무한 반복 실행
  }
}

function reset() {		// 게임 초기화
  targetSize = 30;		// 목표물 크기 초기화
  targetX = random(targetSize/2, width-targetSize/2);		// 목표물 x축 좌표 초기화
  targetY = 0;			// 목표물 y축 좌표 초기화
  targetSpeed = 2;		// 목표물 속도 초기화
  w = 100;			// 플레이어 폭 초기화
  h = 20;			// 플레이어 높이 초기화
  score = 0;			// 점수 초기화
  imageMode(CENTER);	// 이미지 정렬 모드: 이미지의 가운데에서 폭/높이 설정
}
