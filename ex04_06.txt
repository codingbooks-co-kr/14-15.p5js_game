// 좌우 반사하는 원

let x = 0, speed = 1;

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  bounce();			// 반사하는 동작 (함수 호출부)
  display();			// 표시하는 동작 (함수 호출부)
  move();			// 이동하는 동작 (함수 호출부)
}

function bounce() {		// 반사하는 동작 (함수 본체)
  if ( (x < 0) || (x > width) )
    speed *= - 1;
}

function display() {		// 표시하는 동작 (함수 본체)
  ellipse(x, height/2, 30, 30);
  console.log(x);
}

function move() { 		// 이동하는 동작 (함수 본체)
  x += speed;			// x축 속도(부호: 방향성) 누적 → x축 거리 이동
}
