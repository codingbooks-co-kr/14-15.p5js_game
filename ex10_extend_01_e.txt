// millis() 기반 시간 측정: 0.5초마다 원형 생성

let interval, currentTime, previousTime = 0;

function setup() {
  createCanvas(300, 300);
  background(0);
}

function draw() {
  currentTime = millis();			// 현재시간 설정
  interval = currentTime - previousTime; 	// 시간차(interval) = 현재시간 - 이전시간
  if (interval > 500) {				// 500msec마다 if문은 true가 됨
    previousTime = millis();			// 이전시간을 현재시간으로 갱신
    fill(random(width), random(width), random(width), 200);
    ellipse(random(width), random(height), 50, 50);
  }
}
