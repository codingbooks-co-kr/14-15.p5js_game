// 4단계: 랜덤 미로 만들기, 스페이스 키를 누를 때마다 랜덤 미로 생성

let cols, rows, w = 30, maze = [];	// ❶ maze: 1차원 빈 배열의 생성

function setup() {
  createCanvas(630, 630);		// 예) 30(=w:미로단위크기)×21(=cols,rows) = 630
  designMaze();			// cols×rows=21×21 랜덤 미로 디자인
}

function draw() {
  background(0);
  noStroke();					// 사각형 장벽의 외곽선 삭제
  for (let y = 0; y < rows; y++) {		// rows: 세로축(행)은 홀수 개
    for (let x = 0; x < cols; x++) {		// cols: 가로축(열)은 홀수 개
      if (maze[y][x] === 0) fill(0);		// 경로: 갈 수 있는 길은 검정색 사각형
      else fill(200);				// 장벽: 가로막힌 벽은 회색 사각형
      rect(x*w, y*w, w, w);			// 30(→w:미로단위)*21개 = 630(→캔버스)
    }
  }
}

function designMaze() {			// cols×rows=21×21 랜덤 미로 디자인
  cols = int(width / w);			// 예) cols  = 630 / 30 = 21개
  if (cols % 2 === 0) cols -= 1;		// cols가 짝수이면 홀수로 만들기
  rows = int(height / w);			// 예) rows  = 630 / 30 = 21개
  if (rows % 2 === 0) rows -= 1;		// rows가 짝수이면 홀수로 만들기

  for (let y = 0; y < rows; y++) {		// rows: 세로축(행)은 홀수 개
    maze[y] = [];				// ❷ maze: 2차원 배열로 확장
    for (let x = 0; x < cols; x++) {		// cols: 가로축(열)은 홀수 개
      maze[y][x] = 0;				// ❸ maze: 2차원 배열을 ‘0’(=경로)으로 초기화
    }
  }

  for (let y = 1; y < rows; y += 2) {		// 예) y의 홀수 번째 요소(1,3,5,7,...,19)
    for (let x = 1; x < cols; x += 2) {		// 예) x의 홀수 번째 요소(1,3,5,7,...,19)
      maze[y][x] = 1; 				// 현 위치좌표에 장벽 있음(1)
      let direction = int(random(4));		// 상하좌우 랜덤 방향(0,1,2,3) 정하기
      switch (direction) {
        case 0:
          maze[y][x - 1] = 1; 		// 현 위치좌표의 왼쪽에 사각형 장벽 추가
          break;
        case 1:
          maze[y][x + 1] = 1; 		// 현 위치좌표의 오른쪽에 사각형 장벽 추가
          break;
        case 2:
          maze[y - 1][x] = 1; 	// 현 위치좌표의 위쪽에 사각형 장벽 추가
          break;
        case 3:
          maze[y + 1][x] = 1; 	// 현 위치좌표의 아래쪽에 사각형 장벽 추가
          break;
      }
    }
  }
}

function keyPressed() {			// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') designMaze();	// 랜덤 미로 디자인
}
