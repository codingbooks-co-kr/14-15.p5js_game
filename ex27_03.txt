// [본문 코드] 원과 사각형 객체 좌우 반사 (상속 적용 - 미완성)

let c, s;

function setup() {
  createCanvas(300, 300);
  c = new Circle();			// 원형 객체 생성 및 초기화
  s = new Square();			// 사각형 객체 생성 및 초기화
}

function draw() {
  background(0, 10);			// 페이딩 효과 (투명도 10 부여)
  c.display();				// 메소드 호출: 원형 그리기 동작
  c.move();				// 메소드 호출: 원형 좌우 이동 및 반사
  s.display();				// 메소드 호출: 사각형 그리기 동작
  s.move();				// 메소드 호출: 사각형 좌우 이동 및 반사
}

// [클래스 코드]

class Shape {				// 상위 클래스
  constructor() {			// 상위 생성자: 속성 추가 및 초기화
    this.c = color(255);			// 속성: 색상 초기화
    this.x = width/2;			// 속성: x축 위치좌표 초기화
    this.y = height/2;			// 속성: y축 위치좌표 초기화
    this.d = 30;			// 속성: 크기 초기화
    this.speed = 2;			// 속성: 속도 초기화
  }
  move() {				// 상위 메소드: 좌우 이동 및 반사
    this.x += this.speed;		// x축 속도 누적 → x축 거리 이동 (부호: 방향성)
    if (this.x < 0 || this.x > width) {	// 캔버스의 좌우 경계에서 충돌하는 경우
      this.speed *= -1;		// 속도의 방향 반전, 즉 반사
    }
  }
}


class Circle extends Shape {		// 하위 클래스(Circle)로서 상위 클래스(Shape) 상속
  constructor() {			// 하위 생성자
    super();				// 상위 생성자 호출
    this.y = 100;			// 속성 추가 (상속받은 속성 y의 초기값 변경)
    this.speed = 3;			// 속성 추가 (상속받은 속성 speed의 초기값 변경)
  }
  display() {				// 하위 메소드: 원형 그리기 동작
    noStroke();				// 페이딩 효과를 위해 이동체의 외곽선 제거
    fill(this.c, 0, 0);
    ellipse(this.x, this.y, this.d, this.d);
  }
}

class Square extends Shape {		// 하위 클래스(Square)로서 상위 클래스(Shape) 상속
  constructor() {			// 하위 생성자
    super();				// 상위 생성자 호출
    this.y = 200;			// 속성 추가
    this.speed = -3;			// 속성 추가
  }
  display() {				// 하위 메소드: 사각형 그리기 동작
    noStroke();				// 페이딩 효과를 위해 이동체의 외곽선 제거
    fill(0, this.c, 0);
    rect(this.x, this.y, this.d, this.d);
  }
}
