// 완성: 라켓으로 볼을 쳐 벽돌 깨기, 벽돌을 모두 깨면 다음 단계로 이동하고 볼의 속도 증가, 볼을 놓치면 게임 종료, 스페이스 키를 눌러 재시작

let catchSound, clapSound, cosmosImg, brickImg, score, rows, cols, bricks;
let ballX, ballY, ballSize, ballSpeedX, ballSpeedY, playerX, playerY, playerSpeed;
let w, h, speedUp, previousStage, stage = 1;

function preload() {		// setup() 직전에 호출, 외부 파일의 로딩
  clapSound = loadSound('data/claps.mp3');
  catchSound = loadSound('data/catch.mp3');
  cosmosImg = loadImage('data/cosmos.jpg');
  brickImg = loadImage('data/brick.png');
}

function setup() {
  createCanvas(600, 500);
  reset();					// 실행 초기: 게임 초기화
}

function draw() {
  image(cosmosImg, 0, 0, width, height);	// 배경 이미지 표시

  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);			// 문자열 가운데 정렬
  text('Stage: ' + stage + ', Score: ' + score, width/2, 100);		// 단계, 점수 표시
  
  display();					// 2차원 벽돌, 볼 및 라켓 표시
  playerMove();					// 플레이어(라켓) 좌우 이동
  ballMove();					// 볼 이동 및 반사, 벽돌 깨기
  gamePause();					// 게임 종료 또는 단계 이동
}

function display() {				// 2차원 벽돌, 볼 및 라켓 표시
  for (let i = 0; i < rows; i++) {		// 세로축(행) 순차 반복 처리
    for (let j = 0; j < cols; j++) {		// 가로축(열) 순차 반복 처리
      if (bricks[i][j] === 1) {			// 벽돌 있음:1 (없음:0)
        image(brickImg, j*w, i*h, w, h);	// 벽돌 이미지 표시
      }
    }
  }

  noStroke();
  fill(255);
  ellipse(ballX, ballY, ballSize, ballSize);	// 볼 표시
 
  fill(0, 255, 0);
  rectMode(CENTER);			// 사각형의 정가운데를 기준으로 폭과 높이 설정
  rect(playerX, playerY, w, h);		// 플레이어(라켓) 표시
}

function playerMove() {			// 플레이어(라켓) 좌우 이동
  if (keyIsDown(LEFT_ARROW)) {	// 왼쪽 방향키를 누른 경우
    playerX -= playerSpeed;		// x축 속도(-부호: 왼쪽) 누적 → 왼쪽 이동 
  }
  if (keyIsDown(RIGHT_ARROW)) {	// 오른쪽 방향키를 누른 경우
    playerX += playerSpeed;		// x축 속도(+부호: 오른쪽) 누적 → 오른쪽 이동 
  }
  playerX = constrain(playerX, w/2, width-w/2);	// 플레이어의 좌우 이동 제한
}

function ballMove() {			// 볼 이동 및 반사, 벽돌 깨기
  ballX += ballSpeedX;    		// 볼 x축 속도(부호: 방향성) 누적 → x축 거리 이동 
  ballY += ballSpeedY;   		// 볼 y축 속도(부호: 방향성) 누적 → y축 거리 이동
  if (ballX < ballSize/2 || ballX > width-ballSize/2) {
    ballSpeedX *= -1;			// x축 속도의 방향 반전, 즉 반사
  }
  if (ballY < rows*h) {			// 볼이 캔버스 상단의 벽돌 영역에 들어온 경우
    let x = int(ballX / w);
    let y = int(ballY / h);
    if (bricks[y][x] === 1) {		// ❶벽돌이 있는 경우
      catchSound.play();		// 효과음 재생
      bricks[y][x] = 0;			// 벽돌 제거
      ballSpeedY *= -1;		// y축 속도의 방향 반전, 즉 반사
      score++;				// 점수 증가
    } else if (ballY < ballSize/2) {	// ❷벽돌이 없는 경우 (=캔버스 상단 경계에 닿음)
      ballSpeedY *= -1;		// y축 속도의 방향 반전, 즉 반사
    }
  }

  // 플레이어(라켓)로 볼을 친 경우
  if (ballX > playerX-w/2 && ballX < playerX+w/2 && ballY > playerY-(ballSize+h)/2) {
    ballY = playerY - (ballSize + h) / 2;	// ballY 위치는 라켓 상단(→[참고16.1] 참조)
    ballSpeedY *= -1;				// y축 속도의 방향 반전, 즉 반사
  }
}

function gamePause() {			// 게임 종료 또는 단계 이동
  if (score === rows * cols) {		// 벽돌을 모두 깬 경우
    clapSound.play();			// 효과음 재생
    noStroke();
    fill(255);
    text('YOU WIN', width/2, height/2 - 30);
    text('Hit the space key to start next stage', width/2, height/2 + 30);

    stage++;				// 단계 증가
    noLoop();				// draw() 함수의 반복 실행 일시 정지
  }

  if (ballY > height) {			// 볼이 캔버스 바닥을 벗어난 경우
    noStroke();
    fill(255);
    text('GAME OVER', width/2, height/2);

    stage = 1;				// 1단계
    noLoop();				// draw() 함수의 반복 실행 일시 정지
  }
}

function keyPressed() {			// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {			// 스페이스 키를 누르면 게임 초기화, 재실행
    reset();				// 게임 종료 시: 게임 초기화
    loop();				// draw() 함수의 무한 반복 실행
  }
}

function reset() {			// 게임 초기화
  if (stage === 1) {			// 1단계인 경우
    speedUp = 1;			// 볼 속도 초기화
  } else if (stage > 1 && stage !== previousStage) {		// 2단계 이상인 경우
    speedUp *= 1.2;			// 볼 속도 상승
    if (speedUp > 3) speedUp = 3;	// 볼 속도 최대값: 3
  }
  previousStage = stage;		// 2단계 이상에서 동일 단계는 한 번만 처리

  w = 100;				// 플레이어(라켓)의 폭 초기화
  h = 30;				// 플레이어(라켓)의 높이 초기화
  rows = 2;				// 세로축(행) 벽돌 개수 초기화
  cols = int(width / w);		// 가로축(열) 벽돌 개수 초기화

  // 벽돌 배열 만들기
  bricks = [];				// ❶ bricks: 1차원 빈 배열의 생성
  for (let i = 0; i < rows; i++) {
    bricks[i] = [];			// ❷ bricks: 2차원 빈 배열로 확장
    for (let j = 0; j < cols; j++) {
      bricks[i][j] = 1;			// ❸ bricks: 2차원 배열의 초기화
    }
  }

  ballSize = 30;					// 볼의 크기 초기화
  ballX = random(ballSize/2, width-ballSize/2);		// 볼의 x축 좌표 초기화(랜덤)
  ballY = rows * h;					// 볼의 y축 좌표 초기화
  ballSpeedX = random(2, 3) * speedUp;		// 볼의 x축 속도 상승(랜덤)
  ballSpeedY = random(2, 3) * speedUp;		// 볼의 y축 속도 상승(랜덤)
  playerX = width / 2;					// 플레이어(라켓)의 x축 좌표 초기화
  playerY = height - h;				// 플레이어(라켓)의 y축 좌표 초기화
  playerSpeed = 5;					// 플레이어(라켓)의 속도 초기화
  score = 0;						// 점수 초기화
}
