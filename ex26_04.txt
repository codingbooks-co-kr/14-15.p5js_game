// [본문 코드] 원형 객체 다섯 개 좌우 반사

let c = [];		// 변수 선언 및 빈 배열 생성

function setup() {
  createCanvas(300, 300);
  for (let i = 0; i < 5; i++) { 		// 배열요소를 5개 추가, 요소별 객체 생성 및 초기화
    c[i] = new Circle(color(255, 255, 0), 0, i*70, 30, i/2);
  }
}

function draw() {
  background(0, 10);			// 페이딩 효과 (투명도 10 부여)
  for (let i = 0; i < c.length; i++) { 	// c.length(=5)만큼 반복 처리
    c[i].display();			// 배열요소별 객체의 메소드 호출
    c[i].move();				// 배열요소별 객체의 메소드 호출
  }
}

// [클래스 코드]

class Circle {				// 원형 클래스
  constructor(c, x, y, d, speed) {	// 생성자: 속성 추가 및 초기화
    this.c = c;				// 속성: 색상 초기화
    this.x = x;				// 속성: x축 위치좌표 초기화
    this.y = y;				// 속성: y축 위치좌표 초기화
    this.d = d;				// 속성: 크기 초기화
    this.speed = speed;		// 속성: 속도 초기화
  }
  display() {				// 메소드: 원형 표시
    noStroke();				// 페이딩 효과를 위해 이동체의 외곽선 제거
    fill(this.c);
    ellipse(this.x, this.y, this.d, this.d);
  }
  move() {				// 메소드: 원형 이동 및 좌우 반사
    this.x += this.speed;		// x축 속도 누적 → x축 거리 이동 (부호: 방향성)
    if (this.x < 0 || this.x > width) {	// 캔버스의 x축 좌우 경계에서 충돌 시
      this.speed *= -1;		// 속도의 방향 반전, 즉 반사
    }
  }
}
