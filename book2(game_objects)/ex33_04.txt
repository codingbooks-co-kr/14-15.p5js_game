// 완성: 방향키로 플레이어를 좌우로 점프 이동 및 낙하하면서 적을 피해 목적지에 도착하면 다음 단계로 이동. 단계별로 적의 개수 증가, 적과 충돌하면 게임 종료. 스페이스 키를 누르면 재시작

let catchSound, burstSound, cosmosImg, burstImg, platformImg, enemyImg, playerImg;
let player, target, enemies, enemyNum, previousStage, stage = 1;

function preload() {			// setup() 직전에 호출, 외부 파일의 로딩
  catchSound = loadSound('data/catch.mp3');
  burstSound = loadSound('data/burst.mp3');
  cosmosImg = loadImage('data/cosmos.jpg');
  burstImg = loadImage('data/burst.png');
  platformImg = loadImage('data/brick.png');
  enemyImg = loadImage('data/enemy.png');
  playerImg = loadImage('data/rocket2.png');
}

function setup() {
  createCanvas(600, 600);
  reset();				// 실행 초기: 게임 초기화
}

function draw() {
  image(cosmosImg, width/2, height/2, width, height);	// 배경 이미지 표시

  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);		// 문자열 가운데 정렬
  text('Stage: ' + stage, width/2, 30);	// 단계 표시

  platform.display();			// 목적지 객체 표시
  player.display();			// 플레이어 객체 표시
  player.move();			// 플레이어 객체 이동

  if (keyIsDown(LEFT_ARROW)) {	// 왼쪽 방향키를 누른 경우
    player.jump(-1);			// 플레이어 객체 왼쪽 방향 점프
  }
  if (keyIsDown(RIGHT_ARROW)) {	// 오른쪽 방향키를 누른 경우
    player.jump(1);			// 플레이어 객체 오른쪽 방향 점프
  }
  
  if (player.land(platform)) {		// 플레이어가 플랫폼에 착륙한 경우
    fill(255);
    text('YOU WIN', width/2, height/2 - 30);
    text('Hit the space key to start next stage', width/2, height/2 + 30);

    stage++;				// 단계 상승
    noLoop();				// draw() 함수의 반복 실행 일시 정지
  }
  
  for (let i = 0; i < enemies.length; i++) {
    enemies[i].display();		// 적 객체 표시
    enemies[i].move();			// 적 객체 이동
    if (player.hit(enemies[i]) || player.drop()) {		// 적과 충돌하거나 바닥을 벗어남
      player.explode();			// 플레이어가 폭발하는 효과 표시
      fill(255);
      text('GAME OVER', width/2, height/2);

      stage = 1;			// 1단계
      noLoop();				// draw() 함수의 반복 실행 일시 정지
    }
  }
}

function keyPressed() {			// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {			// 스페이스 키를 누르면 게임 초기화, 재실행
    reset();				// 게임 종료 시: 게임 초기화
    loop();				// draw() 함수의 무한 반복 실행
  }
}

function reset() {			// 게임 초기화
  if (stage === 1) {			// 1단계인 경우
    enemyNum = 5;			// 적 개수 초기화
  } else if (stage > 1 && stage !== previousStage) {		// 2단계 이상인 경우
    enemyNum++;			// 적 개수 증가
  }
  previousStage = stage;		// 2단계 이상에서 동일 단계는 한 번만 처리

  platform = new Platform();		// 목적지 객체 생성 및 초기화
  player = new Player();		// 플레이어 객체 생성 및 초기화
  enemies = [];			// 적: 빈 객체 배열로 초기화
  for (let i = 0; i < enemyNum; i++) {
    enemies[i] = new Enemy();	// 적 객체 생성 및 초기화
  }

  textSize(30);
  textAlign(CENTER, CENTER);		// 문자열 가운데 정렬
  imageMode(CENTER);		// 이미지 정렬 모드: 이미지의 가운데에서 폭/높이 설정
}

class Platform {			// 목적지 클래스
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.w = 80;			// 목적지 폭 초기화
    this.h = 30;			// 목적지 높이 초기화
    this.x = random(this.w/2, width-this.w/2);		// 목적지 x축 좌표 초기화(랜덤)
    this.y = height - 50;				// 목적지 y축 좌표 초기화
  }
  display() {						// 메소드: 목적지 객체 표시
    image(platformImg, this.x, this.y, this.w, this.h);	// 목적지 이미지 표시
  }
}

class Enemy {				// 적 클래스
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.size = 40;			// 적 크기 초기화
    this.x = random(this.size/2, width-this.size/2);	// 적 x축 좌표 초기화(랜덤)
    this.y = height - this.size/2;			// 적 y축 좌표 초기화
    this.speedX = random(-2, 2);	// 적 x축 속도 초기화(랜덤)
    this.speedY = random(-2, 2);	// 적 y축 속도 초기화(랜덤)
  }
  display() {				// 메소드: 적 객체 표시
    image(enemyImg, this.x, this.y, this.size, this.size);	// 적 이미지 표시
  }
  move() {				// 메소드: 적 객체 이동
    this.x += this.speedX;   		// x축 속도 누적 → x축 거리 이동
    this.y += this.speedY;   		// y축 속도 누적 → y축 거리 이동
    if (this.x < this.size/2 || this.x > width - this.size/2) {	// x축 경계 반사조건
      this.speedX *= -1;		// 속도의 방향 반전, 즉 반사
    }
    if (this.y < this.size/2 || this.y > height - this.size/2) {	// y축 경계 반사조건
      this.speedY *= -1;		// 속도의 방향 반전, 즉 반사
    }
  }
}

class Player {				// 플레이어 클래스 
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.size = 40;			// 플레이어 크기 초기화
    this.x = width / 2;			// 플레이어 x축 좌표 초기화
    this.y = this.size;			// 플레이어 y축 좌표 초기화
    this.speedX = 0;			// 플레이어 x축 속도 초기화
    this.speedY = 0;			// 플레이어 y축 속도 초기화
    this.gravity = 0;			// 플레이어 중력가속도 초기화
  } 
  display() {				// 메소드: 플레이어 객체 표시
    image(playerImg, this.x, this.y, this.size, this.size);	// 플레이어 이미지 표시
  }
  explode() {				// 메소드: 플레이어 객체 폭발 효과 표시
    burstSound.play();			// 효과음 재생
    image(burstImg, this.x, this.y, this.size*2, this.size*2);	// 폭발 이미지 표시
  }
  move() {				// 메소드: 플레이어 객체 이동
    this.speedY += this.gravity;	// y축 속도에 중력 누적 → y축 가속
    this.x += this.speedX;   		// x축 속도 누적 → x축 거리 이동
    this.y += this.speedY;   		// y축 속도 누적 → y축 거리 이동
    this.x = constrain(this.x, this.size/2, width - this.size/2);		// x축 이동 제한
  }
  jump(turn) {				// 메소드: 플레이어 객체 점프
    this.speedX = 2 * turn;		// 플레이어 x축 속도 (turn: -부호:좌향, +부호:우향)
    this.speedY = -2;			// 플레이어 y축 속도 (-부호: 상향)
    this.gravity = 0.2;			// 플레이어 중력가속도
  }
  land(other) {				// 메소드: 플레이어가 플랫폼에 착륙했는지 여부 판단
    if (this.x > other.x - other.w/2 && this.x < other.x + other.w/2 && this.y > other.y - (this.size + other.h) /2) {	// 9.3절의 if문 참조
      return true;
    } else {
      return false;
    }
  }
  drop() {				// 메소드: 플레이어가 캔버스 바닥을 벗어났는지 여부 판단
    if (player.y > height) return true;	// 플레이어가 캔버스 바닥을 벗어난 경우
    else return false;			// 캔버스 바닥을 벗어나지 않은 경우
  }
  hit(other) {				// 메소드: 두 객체의 충돌 여부 판단
    let distance = dist(this.x, this.y, other.x, other.y);		// 두 객체의 거리 측정
    if (distance < (this.size + other.size) / 2) return true;	// 두 객체의 충돌 시
    else return false;						// 두 객체의 미충돌 시
  }
}
