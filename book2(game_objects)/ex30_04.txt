// 완성: 키보드 방향키로 플레이어를 좌우로 이동하면서 낙하하는 적 피하기, 충돌하면 게임 종료, 스페이스 키를 누르면 재시작

let burstSound, cosmosImg, burstImg, enemyImg, playerImg, player, enemies;
let y1, y2, interval, currentTime, endTime, enemyNum, topInterval, firstStep = true;

function preload() {				// setup() 직전에 호출, 외부 파일의 로딩
  burstSound = loadSound('data/burst.mp3');
  cosmosImg = loadImage('data/cosmos.jpg');
  burstImg = loadImage('data/burst.png');
  enemyImg = loadImage('data/rocket4.png');
  playerImg = loadImage('data/rocket2.png');
}

function setup() {
  createCanvas(400, 600);
  reset();					// 실행 초기: 게임 초기화
}

function draw() {
  image(cosmosImg, 0, y1, width, height); 	// 배경 이미지1 표시(→21.6절 참조)
  image(cosmosImg, 0, y2, width, height); 	// 배경 이미지2 표시(→21.6절 참조)
  y1 += 2;					// 배경 이미지1 아래로 이동
  y2 += 2;					// 배경 이미지2 아래로 이동
  if (y1 > height) y1 = -height;		// 이미지1: 캔버스를 벗어나면 y좌표 재설정
  if (y2 > height) y2 = -height;		// 이미지2: 캔버스를 벗어나면 y좌표 재설정
  
  currentTime = millis() / 1000;		// 게임 시작 후의 경과시간(초) 측정
  interval = currentTime - endTime; 		// 시간차(interval) = 현재시간 - 종료시간
  if (interval > topInterval) topInterval = interval;	// 최장 주행시간 갱신

  noStroke();
  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);			// 문자열 가운데 정렬
  text('Driving Time: ' + nf(interval, 3, 1) + ' sec', width/2, 30);	// 주행시간
  text('Top: ' + nf(topInterval, 3, 1) + ' sec', width/2, 70);		// 최장 주행시간

  player.display();				// 플레이어 객체 표시
  if (keyIsDown(LEFT_ARROW)) {		// 왼쪽 방향키를 누른 경우
    player.move(-1);				// 플레이어 객체 왼쪽 이동
  }
  if (keyIsDown(RIGHT_ARROW)) {		// 오른쪽 방향키를 누른 경우
    player.move(1);				// 플레이어 객체 오른쪽 이동
  }

  for (let i = 0; i < enemies.length; i++) {
    enemies[i].display();			// 적 객체 표시
    enemies[i].move();				// 적 객체 이동
    if (player.hit(enemies[i])) {			// 플레이어∼적 충돌한 경우
      player.explode();				// 폭발하는 효과 표시
      fill(255);
      text('GAME OVER', width/2, height/2);
      noLoop();					// draw() 함수의 반복 실행 일시 정지
    }
  }
}

function keyPressed() {				// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {				// 스페이스 키를 누르면 게임 초기화, 재실행
    reset();					// 게임 종료 시: 게임 초기화
    loop();					// draw() 함수의 무한 반복 실행
  }
}

function reset() {				// 게임 초기화
  y1 = 0;					// 배경 이미지1 y좌표 초기화
  y2 = -height;				// 배경 이미지2 y좌표 초기화
  player = new Player();			// 플레이어 객체 생성 및 초기화
  enemies = [];				// 적: 빈 객체 배열로 초기화
  enemyNum = 8;				// 적 개수 초기화
  let spacing = width / enemyNum;		// 적 객체 사이의 간격
  for (let i = 0; i < enemyNum; i++) {
    enemies[i] = new Enemy(i*spacing+10);	// 적 객체 생성 및 초기화
  }

  if (firstStep) {
    endTime = 0;				// 첫 시작만 endTime을 0으로 초기화
    topInterval = 0;				// 최장 주행시간 초기화
  } else {
    endTime = millis() / 1000;			// 두 번째부터 현재시간(초)으로 초기화
  }
  firstStep = false;
}

class Enemy {				// 적 클래스
  constructor(x) {			// 생성자: 속성 추가 및 초기화
    this.x = x;				// 적 x축 좌표 초기화
    this.y = 0;				// 적 y축 좌표 초기화
    this.w = 30;			// 적 폭 초기화
    this.h = 60;			// 적 높이 초기화
    this.speed = random(1, 6);	// 적 속도 초기화(랜덤)
  }
  display() {				// 메소드: 적 객체 표시
    image(enemyImg , this.x, this.y, this.w, this.h);	// 적 이미지 표시
  }
  move() {				// 메소드: 적 객체 낙하
    this.y += this.speed;		// y축 속도 누적 → y축 거리 이동 (부호: 방향성)
    if (this.y > height) {		// 적 객체가 캔버스 화면 바닥을 벗어나는 경우
      this.y = -this.h;			// 다시 캔버스 화면 위쪽에서 시작
      this.speed = random(1, 6);	// 낙하 속도 초기화(랜덤)
    }
  }
}

class Player {				// 플레이어 클래스 
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.w = 30;			// 플레이어 폭 초기화
    this.h = 60;			// 플레이어 높이 초기화
    this.x = width/2;			// 플레이어 x축 좌표 초기화
    this.y = height-this.h;		// 플레이어 y축 좌표 초기화
    this.speed = 3;			// 플레이어 속도 초기화
  }
  display() {				// 메소드: 플레이어 객체 표시
    image(playerImg, this.x, this.y, this.w, this.h);	// 플레이어 이미지 표시
  }
  explode() {				// 메소드: 플레이어 객체 폭발 효과 표시
    burstSound.play();			// 효과음 재생
    image(burstImg, this.x, this.y, this.w*2, this.h*2);	// 폭발 이미지 표시
  }
  move(turn) {				// 메소드: 플레이어 객체 이동
    this.x += this.speed * turn;	// x축 속도 누적 → x축 거리 이동 (turn: 방향성)
    this.x = constrain(this.x, 0, width-this.w);		// x축 이동 제한
  }
  hit(other) {				// 메소드: 두 사각형의 충돌 여부 판단
    if (this.x < other.x+other.w && this.x+this.w > other.x && this.y < other.y+other.h && this.y+this.h > other.y) {
      return true;
    } else {
      return false;
    }
  }
}
