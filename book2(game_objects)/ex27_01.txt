// [본문 코드] 원과 사각형 객체 좌우 반사 (상속 미적용)

let c;					// (원형 객체를 위한) 변수 선언
let s;					// (사각형 객체를 위한) 변수 선언

function setup() {
  createCanvas(300, 300);
  c = new Circle();			// 원형 객체 생성 및 초기화
  s = new Square();			// 사각형 객체 생성 및 초기화
}

function draw() {
  background(0, 10);			// 페이딩 효과 (투명도 10 부여)
  c.display();				// 원형 객체의 display() 메소드 호출
  c.move();				// 원형 객체의 move() 메소드 호출
  s.display();				// 사각형 객체의 display() 메소드 호출
  s.move();				// 사각형 객체의 move() 메소드 호출
}

// [클래스 코드]

class Circle {				// 원형 클래스 코드
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.c = color(255);			// 속성: 색상 초기화
    this.x = width/2;			// 속성: x축 위치좌표 초기화
    this.y = 100;			// 속성: y축 위치좌표 초기화
    this.d = 30;			// 속성: 크기 초기화
    this.speed = 2;			// 속성: 속도 초기화
  }
  display() {				// 메소드: 그리기 동작
    noStroke();				// 페이딩 효과를 위해 이동체의 외곽선 제거
    fill(this.c);
    ellipse(this.x, this.y, this.d, this.d);
  }
  move() {				// 메소드: 좌우 이동 및 반사 동작
    this.x += this.speed;		// x축 속도 누적 → x축 거리 이동 (부호: 방향성)
    if (this.x < 0 || this.x > width) {	// 캔버스의 좌우 경계에서 충돌하는 경우
      this.speed *= -1;		// 속도의 방향 반전, 즉 반사
    }
  }
}

class Square {				// 사각형 클래스 코드
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.c = color(255);			// 속성: 색상 초기화
    this.x = width/2;			// 속성: x축 위치좌표 초기화
    this.y = 200;			// 속성: y축 위치좌표 초기화
    this.d = 30;			// 속성: 크기 초기화
    this.speed = -2;			// 속성: 속도 초기화
  }
  display() {				// 메소드: 그리기 동작
    noStroke();				// 페이딩 효과를 위해 이동체의 외곽선 제거
    fill(this.c);
    rect(this.x, this.y, this.d, this.d);
  }
  move() {				// 메소드: 좌우 이동 및 반사 동작
    this.x += this.speed;		// x축 속도 누적 → x축 거리 이동 (부호: 방향성)
    if (this.x < 0 || this.x > width) {	// 캔버스의 좌우 경계에서 충돌하는 경우
      this.speed *= -1;		// 속도의 방향 반전, 즉 반사
    }
  }
}
