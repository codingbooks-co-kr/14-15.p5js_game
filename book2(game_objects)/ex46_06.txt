// DC모터 정방향, 역방향 회전 및 정지를 반복

#define DCMOTOR_FWD 9		// 모터의 정방향 제어핀
#define DCMOTOR_BWD 10		// 모터의 역방향 제어핀

void setup() {
  pinMode(DCMOTOR_FWD, OUTPUT);	// 모터의 정방향 제어핀을 출력으로 사용
  pinMode(DCMOTOR_BWD, OUTPUT);	// 모터의 역방향 제어핀을 출력으로 사용
}

void loop() {
  digitalWrite(DCMOTOR_FWD, 1);    	// 정방향 회전
  digitalWrite(DCMOTOR_BWD, 0);
  delay(2000);				// 2초
  digitalWrite(DCMOTOR_FWD, 0);   	// 정지
  digitalWrite(DCMOTOR_BWD, 0);
  delay(2000);				// 2초
  digitalWrite(DCMOTOR_FWD, 0);   	// 역방향 회전
  digitalWrite(DCMOTOR_BWD, 1);
  delay(2000);				// 2초
  digitalWrite(DCMOTOR_FWD, 0);   	// 정지
  digitalWrite(DCMOTOR_BWD, 0);
  delay(2000);				// 2초
}
