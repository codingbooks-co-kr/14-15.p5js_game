// 1단계: 10개의 원형 목표물 객체들이 상하좌우 반사

let targets = [];				// (객체 배열을 위한) 변수 선언 및 빈 배열 생성

function setup() {
  createCanvas(600, 600);
  for (let i = 0; i < 10; i++) {
    targets[i] = new Target();		// 목표물 객체 생성 및 초기화
  }
}

function draw() {
  background(0, 10);			// 페이딩 효과 (투명도 10 부여)
  for (let i = 0; i < targets.length; i++) {
    targets[i].display();			// 목표물 객체 표시
    targets[i].move();			// 목표물 객체 이동
  }
}

class Target {				// 목표물 클래스
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.size = random(20, 60);			// 크기 초기화(랜덤)
    this.x = random(this.size/2, width-this.size/2);	// x축 좌표 초기화(랜덤)
    this.y = random(this.size/2, height-this.size/2);	// y축 좌표 초기화(랜덤)
    this.speedX = random(-3, 3);			// x축 속도 초기화(랜덤)
    this.speedY = random(-3, 3);			// y축 속도 초기화(랜덤)
  }
  display() {				// 메소드: 목표물 객체 표시
    noStroke();				// 부드러운 페이딩 효과를 위해 이동체의 외곽선 제거
    fill(255, 200);
    ellipse(this.x, this.y, this.size, this.size);		// 목표물 표시
  }
  move() {				// 메소드: 목표물 객체 이동 및 반사
    this.x += this.speedX;   		// x축 속도 누적 → x축 거리 이동 (부호: 방향성)
    this.y += this.speedY;   		// y축 속도 누적 → y축 거리 이동 (부호: 방향성)
    if (this.x < this.size/2 || this.x > width - this.size/2) {	// x축 경계 반사조건
      this.speedX *= -1;		// 속도의 방향 반전, 즉 반사
    }
    if (this.y < this.size/2 || this.y > height - this.size/2) {	// y축 경계 반사조건
      this.speedY *= -1;		// 속도의 방향 반전, 즉 반사
    }
  }
}
