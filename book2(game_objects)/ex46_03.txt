// RGB LED를 Red→Green→Blue 순으로 점멸 반복

#define RedLED 11
#define GreenLED 5
#define BlueLED 3

void setup() {
  pinMode(RedLED, OUTPUT);		// RedLED핀을 출력으로 사용
  pinMode(GreenLED, OUTPUT);	// GreenLED핀을 출력으로 사용
  pinMode(BlueLED, OUTPUT);		// BlueLED핀을 출력으로 사용
}

void loop() {
  // Red색: (R, G, B) = (1, 0, 0)
  digitalWrite(RedLED, 1);
  digitalWrite(GreenLED, 0);
  digitalWrite(BlueLED, 0);
  delay(1000);

  // Green색: (R, G, B) = (0, 1, 0)
  digitalWrite(RedLED, 0);
  digitalWrite(GreenLED, 1);
  digitalWrite(BlueLED, 0);
  delay(1000);

  // Blue색: (R, G, B) = (0, 0, 1)
  digitalWrite(RedLED, 0);
  digitalWrite(GreenLED, 0);
  digitalWrite(BlueLED, 1);
  delay(1000);
}
