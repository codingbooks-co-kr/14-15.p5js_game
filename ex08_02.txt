// 캔버스상에 이미지 출력

let imageFile;					// ❶ (이미지 객체) 변수 선언

function preload() {				// setup() 직전에 호출, 외부 파일의 로딩
  imageFile = loadImage('data/flower.jpg');	// ❷ 이미지 파일 읽고 객체(imageFile) 생성
}

function setup() {
  createCanvas(imageFile.width, imageFile.height);	// 캔버스 크기를 이미지의 크기로 설정
  console.log(imageFile);
}

function draw() {
  background(0);
  image(imageFile, 0, 0);			// ❸ 캔버스의 (0, 0) 좌표에 이미지 표시
}
