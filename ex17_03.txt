// 3단계: 목표물을 획득할수록 점수, 플레이어의 이동 속도 및 적의 추적 속도 증가 (키보드 방향키로 플레이어 이동)

let targetX, targetY, enemyX, enemyY, playerX, playerY, score = 0;
let targetSize = 30, enemySize = 30, playerSize = 30, enemySpeed = 1, playerSpeed = 2;

function setup() {
  createCanvas(600, 600);
  targetX = random(targetSize, width-targetSize);	// 목표물의 x축 좌표 초기화(랜덤)
  targetY = random(targetSize, height-targetSize);	// 목표물의 y축 좌표 초기화(랜덤)
  enemyX = width;					// 적의 x축 좌표 초기화
  enemyY = height;					// 적의 y축 좌표 초기화
  playerX = width / 2;					// 플레이어의 x축 좌표 초기화
  playerY = height / 2;				// 플레이어의 y축 좌표 초기화
}

function draw() {
  background(0, 10);			// 페이딩 효과 (투명도 10 부여)
  
  noStroke();				// 부드러운 페이딩 효과를 위해 이동체의 외곽선 제거
  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);				// 문자열 가운데 정렬
  text('Score: ' + score, width/2, 30);			// 점수 표시

  fill(255);
  ellipse(targetX, targetY, targetSize, targetSize);	// 목표물 표시

  fill(255, 0, 0);
  ellipse(enemyX, enemyY, enemySize, enemySize);	// 적 표시

  fill(0, 255, 0);
  ellipse(playerX, playerY, playerSize, playerSize);	// 플레이어 표시

  if (keyIsDown(LEFT_ARROW)) {	// 왼쪽 방향키를 누른 경우
    playerX -= playerSpeed;		// x축 속도(-부호: 왼쪽) 누적 → 왼쪽 이동 
  }
  if (keyIsDown(RIGHT_ARROW)) {	// 오른쪽 방향키를 누른 경우
    playerX += playerSpeed;		// x축 속도(+부호: 오른쪽) 누적 → 오른쪽 이동 
  }
  if (keyIsDown(UP_ARROW)) {	// 위쪽 방향키를 누른 경우
    playerY -= playerSpeed;		// y축 속도(-부호: 위쪽) 누적 → 위쪽 이동 
  }
  if (keyIsDown(DOWN_ARROW)) {	// 아래쪽 방향키를 누른 경우
    playerY += playerSpeed;		// y축 속도(+부호: 아래쪽) 누적 → 아래쪽 이동 
  }
  playerX = constrain(playerX, playerSize/2, width-playerSize/2);	// 좌우 이동 제한
  playerY = constrain(playerY, playerSize/2, height-playerSize/2);	// 상하 이동 제한

  let distance = dist(enemyX, enemyY, playerX, playerY);		// 적∼플레이어 거리
  if (distance < (enemySize + playerSize) / 2) {			// 적∼플레이어 충돌
    fill(255);
    text('GAME OVER', width/2, height/2);
    noLoop();					// draw() 함수의 반복 실행 일시 정지
  }

  let dx = (playerX - enemyX) / distance;	// dx가 +이면 playerX가 오른쪽 위치(-:반대)
  let dy = (playerY - enemyY) / distance;	// dy가 +이면 playerY가 아래쪽 위치(-:반대)
  dx *= enemySpeed;				// dx 속도 증가
  dy *= enemySpeed;				// dy 속도 증가
  enemyX += dx;				// dx가 +이면 오른쪽 이동, -이면 왼쪽 이동
  enemyY += dy;				// dy가 +이면 아래쪽 이동, -이면 위쪽 이동

  distance = dist(targetX, targetY, playerX, playerY);	// 목표물∼플레이어 거리
  if (distance < (targetSize + playerSize) / 2) {	// 목표물과 플레이어 충돌
    targetX = random(targetSize, width-targetSize);	// 목표물의 x축 좌표 랜덤 변경
    targetY = random(targetSize, height-targetSize);	// 목표물의 y축 좌표 랜덤 변경
    enemySpeed += 0.3;			// 적 속도 증가
    playerSpeed++;				// 플레이어 속도 증가
    score++;					// 점수 증가
  }
}
