// 2단계: 마우스를 누를 때마다 랜덤/정지를 반복하고 정지되었을 때 랜덤 주사위값(1∼6) 출력
// 주사위 유니코드 문자: https://kr.piliapp.com/symbol/dice/ 

let dice = ['⚀', '⚁', '⚂', '⚃', '⚄', '⚅'];		// 주사위 유니코드 문자 배열
let index = 0, stop = false;

function setup() {
  createCanvas(400, 200);
}

function draw() {
  background(0);

  fill(255);
  textSize(100);
  textAlign(CENTER, CENTER);			// 문자열 가운데 정렬
  text(dice[index], width/2, height/2);		// 주사위 유니코드 문자 배열요소 출력

  textSize(30);
  text(index+1, width/2, 160);			// 주사위값(1∼6) 출력

  index = int(random(dice.length));		// index = int(random(6)) = 0∼5의 정수
}

function mousePressed() {			// 이벤트함수: 마우스를 누르면 한 번 실행
  stop = !stop;					// stop값은 true 또는 false로 토글
  if (stop) noLoop();				// draw() 함수의 반복 실행 일시 정지
  else loop();					// draw() 함수의 무한 반복 실행
}
