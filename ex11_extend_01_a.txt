// ❶ 막대형 타이머 구현

let timer = 0, setTime = 10;		// setTime: 10초

function setup() {
  createCanvas(500, 200);
}

function draw() {
  background(0);
  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);			// 문자열 가운데 정렬
  text('Timer: ' + nf(timer, 2, 1), width/2, 40);

  // 타이머 전체 길이를 사각형 폭(0∼400)으로 표현
  fill(255);
  rect(50, height/2-20, 400, 40);

  // 막대형 타이머의 진행바 표현
  let pos = map(timer, 0, setTime, 0, 400);	// 초(0∼10)를 폭(0∼400)으로 범위 변환
  fill(255, 0, 0);
  rect(50, height/2-20, pos, 40);		// 현재 진행값(pos)을 사각형 폭으로 표시
  
  if (timer < setTime) {	// 10초 되기 전 timer 누적
    timer += 1/60;		// draw() 초당 60회 실행 → timer: 초당 (1/60)*60회≈1초
  } else {			// 10초가 되면 타이머 정지
    fill(255);
    text('GAME OVER', width/2, height/2);
    noLoop();			// draw() 함수의 반복 실행 일시 정지
  }
}

function keyPressed() {		// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {		// 스페이스 키를 누르면 타이머 초기화, 재실행
    timer = 0;			// 타이머 초기화
    loop();			// draw() 함수의 무한 반복 실행
  }
}
