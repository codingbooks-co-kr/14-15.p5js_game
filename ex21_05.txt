// 5단계: 방향키로 플레이어 좌우 이동, 레이저빔을 발사하여 낙하하는 목표물 맞추기

let player, targets = [], shoot = false;

function setup() {
  createCanvas(500, 600);
  player = { x: width/2, y: height-30, size: 30, speed: 3 };	  // 플레이어 객체 초기화
}

function draw() {
  background(0);
  
  noStroke();				// 레이저 발사 시 외곽선 색상에 영향을 끼치지 않음
  fill(0, 255, 0);
  ellipse(player.x, player.y, player.size, player.size);		  // 플레이어 객체 표시
  
  if (keyIsDown(LEFT_ARROW)) {	// 왼쪽 방향키를 누른 경우
    player.x -= player.speed;		// x축 속도(-부호: 왼쪽) 누적 → 왼쪽 이동 
  }
  if (keyIsDown(RIGHT_ARROW)) {	// 오른쪽 방향키를 누른 경우
    player.x += player.speed;		// x축 속도(+부호: 오른쪽) 누적 → 오른쪽 이동 
  }
  player.x = constrain(player.x, 0, width);		// 플레이어 좌우 이동 제한

  if (frameCount % 30 === 0) {	// 0.5초마다 target 객체 생성 후 targets 배열에 추가
    let target = { x: random(width), y: 0, size: random(20, 40), speed: random(1, 2) };
    targets.push(target);		// 생성된 target 객체를 targets 배열에 추가
  }

  for (let i = targets.length-1; i >= 0; i--) {		// 목표물 객체별 역순 반복 처리
    fill(255, 200);
    ellipse(targets[i].x, targets[i].y, targets[i].size, targets[i].size);	// 목표물 표시

    targets[i].y += targets[i].speed;	// 목표물의 y축 속도 누적 → 아래쪽 이동
    if (targets[i].y > height) {		// 목표물이 캔버스 바닥을 벗어난 경우
      targets.splice(i, 1);		// ❶targets[i] 목표물 객체 제거
    }
  }

  if (shoot) {				// shoot = true일 때 레이저빔 발사
    let laser = player.x;		// 레이저빔 발사 시 x축 좌표
    shoot = false;			// draw()의 반복 처리 시 레이저빔은 한 번만 발사 

    stroke(255, 0, 0);
    strokeWeight(3);
    line(laser, player.y, laser, 0);	// 레이저빔 표시
    
    for (let i = targets.length-1; i >= 0; i--) {		// 목표물 객체별 역순 반복 처리
      if (laser > (targets[i].x - targets[i].size/2) && laser < (targets[i].x + targets[i].size/2)) {
        noStroke();
        fill(255, 0, 0, 200);
        ellipse(targets[i].x, targets[i].y, targets[i].size*2, targets[i].size*2);	// 폭발
        targets.splice(i, 1);		// ❷targets[i] 요소 제거
      }
    }
  }
}

function keyPressed() {			// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') shoot = true;	// 스페이스 키를 누르면 레이저빔 한 번 발사
}
