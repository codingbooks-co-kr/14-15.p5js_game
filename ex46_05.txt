// 부저로 “도미솔” 음 출력

#define BUZZER 13

void setup() {
  pinMode(BUZZER, OUTPUT);		// 부저핀을 출력으로 사용
  tone(BUZZER, 262, 500);		// “도”
  delay(1000);
  noTone(BUZZER);			// 무음
  delay(1000);
  tone(BUZZER, 330, 500);		// “미”
  delay(1000);
  tone(BUZZER, 392, 500);		// “솔”
  delay(1000);
}

void loop() { }
