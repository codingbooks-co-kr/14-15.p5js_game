// 완성: 마우스쪽으로 플레이어를 회전하면서 마우스를 누르는 동안 총알을 발사하여 추적하는 적을 소멸. 7점을 획득하면 다음 단계로 이동하고 적 생성율 증가, 적과 충돌하면 게임 종료, 스페이스 키를 누르면 재시작

let clapSound, burstSound, cosmosImg, burstImg, enemyImg, playerImg, score;
let player, enemies, bullets, rate, previousStage, stage = 1;

function preload() {				// setup() 직전에 호출, 외부 파일의 로딩
  clapSound = loadSound('data/claps.mp3');
  burstSound = loadSound('data/burst.mp3');
  cosmosImg = loadImage('data/cosmos.jpg');
  burstImg = loadImage('data/burst.png');
  enemyImg = loadImage('data/enemy.png');
  playerImg = loadImage('data/rocket1.png');
}

function setup() {
  createCanvas(600, 600);
  reset();					// 실행 초기: 게임 초기화
}

function draw() {
  image(cosmosImg, width/2, height/2, width, height);	// 배경 이미지 표시

  fill(255);
  text('Stage: ' + stage + ', Score: ' + score, width/2, 30);

  player.display();				// 플레이어 객체 표시
  player.move();				// 플레이어 객체 이동

  if (mouseIsPressed) {			// 마우스를 누른 상태에서
    if (frameCount % 5 === 0) {		// 5/60초마다 총알 생성
      let bullet = new Bullet(player.angle);	// 총알 객체 생성 및 초기화
      bullets.push(bullet);			// bullets 배열 끝에 bullet 객체 추가
    }
  }
  
  if (random(1) < rate) {		// 매 프레임 적 객체의 생성율(rate: 0.01∼0.1)
    let enemy = new Enemy();	// 적 객체 생성 및 초기화
    enemies.push(enemy);		// enemies 배열 끝에 enemy 객체 추가
  }

  for (let i = 0; i < enemies.length; i++) {
    enemies[i].display();		// 적 객체 표시
    enemies[i].move(player);		// 적 객체 플레이어 추적
    if (player.hit(enemies[i])) {		// 플레이어∼적 충돌한 경우
      player.explode();			// 플레이어 폭발하는 효과 표시
      fill(255);
      text('GAME OVER', width/2, height/2);

      stage = 1;			// 1단계
      noLoop();				// draw() 함수의 반복 실행 일시 정지
    }
  }
  
  for (let i = bullets.length-1; i >= 0; i--) {		// 역순 반복 처리: [참고20.1] 참조
    bullets[i].display();			// 총알 객체 표시
    bullets[i].move();			// 총알 객체 이동(=발사)
    if (bullets[i].offCanvas()) {		// 총알 객체가 캔버스에서 사라지면 배열에서 제거
      bullets.splice(i, 1);		// 총알 객체 bullets[i] 삭제
    } else {
      for (let j = enemies.length-1; j >= 0; j--) {	// 역순 반복 처리: [참고20.1] 참조
        if (bullets[i].hit(enemies[j])) {	// 총알∼적 충돌한 경우
          score++;			// 점수 증가
          enemies[j].explode();		// 적 폭발하는 효과 표시
          enemies.splice(j, 1);		// 적 객체 enemies[i] 삭제
          bullets.splice(i, 1);		// 총알 객체 bullets[i] 삭제
          break;			// 반복문 빠져나옴
        }
      }
    }
  }

  if (score >= 7) {			// 점수가 최소 7점인 경우 단계 이동
    clapSound.play();			// 효과음 재생
    fill(255);
    text('YOU WIN', width/2, height/2 - 30);
    text('Hit the space key to start next stage', width/2, height/2 + 30);

    stage++;				// 단계 증가
    noLoop();				// draw() 함수의 반복 실행 일시 정지
  }
}

function keyPressed() {			// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {			// 스페이스 키를 누르면 게임 초기화, 재실행
    reset();				// 게임 종료 시: 게임 초기화
    loop();				// draw() 함수의 무한 반복 실행
  }
}

function reset() {			// 게임 초기화
  if (stage === 1) {			// 1단계인 경우
    rate = 0.01;			// 적 객체의 생성율 초기화
  } else if (stage > 1 && stage !== previousStage) {		// 2단계 이상인 경우
    rate += 0.01;			// 적 객체의 생성율 증가
    if (rate > 0.1) rate = 0.1;		// 적 객체의 생성율 최대치 제한
  }
  previousStage = stage;		// 2단계 이상에서 동일 단계는 한 번만 처리

  player = new Player();		// 플레이어 객체 생성 및 초기화
  enemies = [];			// 적: 빈 객체 배열로 초기화
  bullets = [];				// 총알: 빈 객체 배열로 초기화
  score = 0;				// 점수 초기화

  textSize(30);
  textAlign(CENTER, CENTER);		// 문자열 가운데 정렬
  imageMode(CENTER);		// 이미지 정렬 모드: 이미지의 가운데에서 폭/높이 설정
}

class Enemy {				// 적 클래스
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.x = random(width);		// 적 x축 좌표 초기화(랜덤)
    this.y = random(height);		// 적 y축 좌표 초기화(랜덤)
    this.size = random(20, 60);	// 적 크기 초기화(랜덤)
    this.speed = random(0.5, 1.5);	// 적 속도 초기화(랜덤)
  }
  display() {				// 메소드: 적 객체 표시
    image(enemyImg, this.x, this.y, this.size, this.size);	// 적 이미지 표시
  }
  explode() {				// 메소드: 적 객체 폭발 효과 표시
    burstSound.play();			// 효과음 재생
    image(burstImg, this.x, this.y, this.size*2, this.size*2);	// 폭발 이미지 표시
  }
  move(player) {			// 메소드: 적 객체의 플레이어 추적(→17.1절 참조)
    let distance = dist(this.x, this.y, player.x, player.y);	// 적∼플레이어 거리
    let dx = (player.x - this.x) / distance;	// dx가 +이면 player가 오른쪽 위치(-:반대)
    let dy = (player.y - this.y) / distance;	// dy가 +이면 player가 아래쪽 위치(-:반대)
    dx *= this.speed;			// 적 x축 속도 
    dy *= this.speed;			// 적 y축 속도 
    this.x += dx;			// dx가 +이면 오른쪽 이동, -이면 왼쪽 이동
    this.y += dy; 			// dy가 +이면 아래쪽 이동, -이면 위쪽 이동
  }
}

class Player {				// 플레이어 클래스 
  constructor() {			// 생성자: 속성 추가 및 초기화
    this.x = width / 2;			// 플레이어 x축 좌표 초기화
    this.y = height / 2;		// 플레이어 y축 좌표 초기화
    this.size = 40;			// 플레이어 크기 초기화
    this.angle = 0;			// 플레이어 회전각 초기화
  }
  display() {				// 메소드: 플레이어 객체 표시
    push(); 				// [좌표 변환 설정 시작]
    translate(this.x, this.y);		// 좌표 이동
    rotate(this.angle);			// 좌표 회전
    image(playerImg, 0, 0, this.size, this.size);	// 플레이어 이미지 표시
    pop();				// [좌표 변환 설정 종료]
  }
  explode() {				// 메소드: 플레이어 객체 폭발 효과 표시
    burstSound.play();			// 효과음 재생
    push();
    translate(this.x, this.y);		// 좌표 이동
    image(burstImg, 0, 0, this.size*2, this.size*2);	// 폭발 이미지 표시
    pop();
  }
  move() {				// 메소드: 플레이어 객체의 회전
    let dx = mouseX - this.x;		// 마우스와 플레이어 사이의 x축 좌표값 차분 
    let dy = mouseY - this.y;		// 마우스와 플레이어 사이의 y축 좌표값 차분 
    this.angle = atan2(dy, dx);		// 회전 각도 계산(→[심화35.1] 참조)
  }
  hit(other) {				// 메소드: 두 객체의 충돌 여부 판단
    let distance = dist(this.x, this.y, other.x, other.y);		// 두 객체의 거리 측정
    if (distance < (this.size + other.size) / 2) return true;	// 두 객체의 충돌 시
    else return false;						// 두 객체의 미충돌 시
  }
}

class Bullet {				// 총알 클래스 
  constructor(angle) {			// 생성자: 속성 추가 및 초기화
    this.x = width / 2;			// 총알 x축 좌표 초기화
    this.y = height / 2;		// 총알 y축 좌표 초기화
    this.size = 5;			// 총알 크기 초기화
    this.speedX = 10 * cos(angle);	// angle 방향에서 총알 x축 속도(10) 초기화(→13.1절)
    this.speedY = 10 * sin(angle);	// angle 방향에서 총알 y축 속도(10) 초기화(→13.1절)
  }
  display() {				// 메소드: 총알 객체 표시
    fill(255);
    ellipse(this.x, this.y, this.size, this.size);
  }
  move() {				// 메소드: 총알 객체 이동(=발사)
    this.x += this.speedX;   		// x축 속도 누적 → x축 거리 이동
    this.y += this.speedY;   		// y축 속도 누적 → y축 거리 이동
  }
  hit(other) {				// 메소드: 두 객체의 충돌 여부 판단
    let distance = dist(this.x, this.y, other.x, other.y);		// 두 객체의 거리 측정
    if (distance < (this.size + other.size) / 2) return true;	// 두 객체의 충돌 시
    else return false;						// 두 객체의 미충돌 시
  }
  offCanvas() {				// 메소드: 총알이 캔버스를 벗어나는지 여부 검사
    if (this.x < 0 || this.x > width) return true;	// 캔버스의 좌우경계를 벗어난 경우
    if (this.y < 0 || this.y > height) return true;	// 캔버스의 상하경계를 벗어난 경우
    return false;
  }
}
