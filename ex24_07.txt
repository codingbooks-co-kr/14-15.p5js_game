// 완성: 미로를 탈출하여 도착지에 도착하면 게임 종료되고 경과시간 표시. 스페이스키로 재시작

let clapSound, cosmosImg, brickImg, targetImg, playerImg, playerImgs = [];
let targetX, targetY, playerX, playerY, interval, currentTime, previousTime;
let maze, cols, rows, w, arrived, direction, topInterval, stage = 1;

function preload() {			// setup() 직전에 호출, 외부 파일의 로딩
  clapSound = loadSound('data/claps.mp3');
  cosmosImg = loadImage('data/cosmos.jpg');
  brickImg = loadImage('data/brick.png');
  targetImg = loadImage('data/circle2.png');

  playerImgs[0] = loadImage('data/rocket0.png');	// 이미지0을 playerImgs[0]에 저장
  playerImgs[1] = loadImage('data/rocket1.png');	// 이미지1을 playerImgs[1]에 저장
  playerImgs[2] = loadImage('data/rocket2.png');	// 이미지2를 playerImgs[2]에 저장
  playerImgs[3] = loadImage('data/rocket3.png');	// 이미지3을 playerImgs[3]에 저장
}

function setup() {
  createCanvas(630, 630);		// 예) 30(=w:미로단위크기)×21(=cols,rows) = 630
  reset();				// 실행 초기: 게임 초기화
}

function draw() {
  image(cosmosImg, 0, 0, width, height);	// 배경 이미지 표시

  timerProcess();				// 경과 시간 측정
  display();					// 장벽, 플레이어, 목표물 표시
  gamePause();					// 목적지 도착 시 경과시간 표시, 게임 정지
}

function timerProcess() {			// 경과 시간 측정
  currentTime = millis() / 1000;		// 게임 시작 후의 경과시간(초) 측정
  interval = currentTime - previousTime; 	// 시간차(interval) = 현재시간 - 이전시간
}

function display() {				// 장벽, 플레이어, 목표물 표시
  for (let y = 0; y < rows; y++) {		// rows: 세로축(행)은 홀수 개
    for (let x = 0; x < cols; x++) {		// cols: 가로축(열)은 홀수 개
      if (maze[y][x] === 1) {			// 장벽인 경우
        image(brickImg, x*w, y*w, w, w);	// 장벽 이미지 표시
      }
    }
  }
  image(playerImgs[direction], playerX*w, playerY*w, w, w);	// 플레이어 이미지 표시
  image(targetImg, targetX*w, targetY*w, w, w);		// 목표물 이미지 표시
}

function gamePause() { 	// 목적지 도착 시 경과시간 표시 및 게임 일시 정지
  if (arrived) {			// 목적지에 도착한 경우
    clapSound.play();		// 효과음 재생

    if (stage === 2 || topInterval > interval) {		// 2단계이거나 주행시간이 짧아지면
      topInterval = interval;				// 최단 주행시간 갱신
    }

    fill(255, 255, 0);
    textSize(50);
    textAlign(CENTER, CENTER);		// 문자열 가운데 정렬
    text('Just Arrived', width/2, height/2 - 25);
    text('Time: ' + nf(interval, 2, 1) + 'sec', width/2, height/2 + 25);		// 경과시간
    text('Top: ' + nf(topInterval, 2, 1) + ' sec', width/2, height/2 + 75);	// 최단시간
    noLoop();					// draw() 함수의 반복 실행 일시 정지
  }
}

function keyPressed() {				// 이벤트함수: 자판키를 누르면 한 번 실행
  if (key === ' ') {				// 스페이스 키를 눌러 재시작
    reset();					// 게임 종료 시: 게임 초기화
    loop();					// draw() 함수의 무한 반복 실행
  }

  let x = playerX;				// 플레이어의 x단위좌표값(0∼cols-1) 저장
  let y = playerY;				// 플레이어의 y단위좌표값(0∼rows-1) 저장

  if (keyCode === LEFT_ARROW) {	 	// 왼쪽 키를 누른 경우
    direction = 0;				// 플레이어 이미지: playerImgs[0] 선택
    x--;
  }
  if (keyCode === RIGHT_ARROW) {	 	// 오른쪽 키를 누른 경우
    direction = 1;				// 플레이어 이미지: playerImgs[1] 선택
    x++;
  }
  if (keyCode === UP_ARROW) {	 	// 위쪽 키를 누른 경우
    direction = 2;				// 플레이어 이미지: playerImgs[2] 선택
    y--;
  }
  if (keyCode === DOWN_ARROW) {	 	// 아래쪽 키를 누른 경우
    direction = 3;				// 플레이어 이미지: playerImgs[3] 선택
    y++;
  }

  x = constrain(x, 0, cols-1);			// 플레이어의 좌우 이동 제한: 0∼cols-1
  y = constrain(y, 0, rows-1);			// 플레이어의 상하 이동 제한: 0∼rows-1

  if (maze[y][x] === 0) {			// 이동할 수 있는 경로(=검정색)인 경우
    playerX = x;				// 플레이어의 x증감분 반영
    playerY = y;				// 플레이어의 y증감분 반영
  }

  if (playerX === cols-1 && playerY === rows-1) {		// 목적지에 도착한 경우
    arrived = true;
    stage++;					// 단계 증가
  }
}

function reset() {				// 게임 초기화
  w = 30;					// 미로의 경로/장벽 단위 크기
  cols = int(width / w);			// 예) cols  = 630 / 30 = 21개
  if (cols % 2 === 0) cols -= 1;		// cols가 짝수이면 홀수로 만들기
  rows = int(height / w);			// 예) rows  = 630 / 30 = 21개
  if (rows % 2 === 0) rows -= 1;		// rows가 짝수이면 홀수로 만들기

  designMaze();				// cols×rows=21×21 랜덤 미로 디자인

  targetX = cols - 1;				// 목표물(목적지) x축 좌표 초기화
  targetY = rows - 1;				// 목표물(목적지) y축 좌표 초기화
  playerX = 0;					// 플레이어 x축 좌표 초기화
  playerY = 0;					// 플레이어 y축 좌표 초기화
  direction = 2;				// 플레이어 이미지 방향(0∼3)
  arrived = false;				// 목적지 도착 여부 초기화

  if (stage === 1) {				// 첫 번째 단계 시
    previousTime = 0;				// previousTime(이전시간)은 0으로 초기화
    topInterval = 0;				// 최단 주행시간 초기화
  } else {					// 두 번째 단계부터
    previousTime = millis() / 1000;		// previousTime은 현재시간(초)으로 초기화
  }
}

function designMaze() {			// cols×rows=21×21 랜덤 미로 디자인
  maze = [];					// ❶ maze: 1차원 빈 배열의 생성
  for (let y = 0; y < rows; y++) {		// rows: 세로축(행)은 홀수 개
    maze[y] = [];				// ❷ maze: 2차원 빈 배열로 확장
    for (let x = 0; x < cols; x++) {		// cols: 가로축(열)은 홀수 개
      maze[y][x] = 0;				// ❸ maze: 2차원 배열을 ‘0’(=경로)으로 초기화
    }
  }

  for (let y = 1; y < rows; y += 2) {		// 예) y의 홀수 번째 요소(1,3,5,7,...,19)
    for (let x = 1; x < cols; x += 2) {		// 예) x의 홀수 번째 요소(1,3,5,7,...,19)
      maze[y][x] = 1; 				// 현 위치좌표에 장벽 있음(1)
      let direction = int(random(4));		// 상하좌우 랜덤 방향(0,1,2,3) 정하기
      switch (direction) {
        case 0:
          maze[y][x - 1] = 1; 		// 현 위치좌표의 왼쪽에 사각형 장벽 추가
          break;
        case 1:
          maze[y][x + 1] = 1; 		// 현 위치좌표의 오른쪽에 사각형 장벽 추가
          break;
        case 2:
          maze[y - 1][x] = 1; 		// 현 위치좌표의 위쪽에 사각형 장벽 추가
          break;
        case 3:
          maze[y + 1][x] = 1; 		// 현 위치좌표의 아래쪽에 사각형 장벽 추가
          break;
      }
    }
  }
}
