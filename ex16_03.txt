// 3단계: 마우스로 플레이어(라켓)를 이동하여 볼 치기, 볼을 놓치면 게임 종료 

let playerX, playerY, ballX, ballY, speedX, speedY;
let w = 100, h = 30, ballSize = 30, gravity = 0.3, friction = 0.95;

function setup() {
  createCanvas(600, 500);
  ballX = random(ballSize/2, width-ballSize/2);		// 볼 x축 좌표 초기화(랜덤)
  ballY = ballSize / 2;					// 볼 y축 좌표 초기화
  speedX = random(2, 5);				// 볼 x축 속도 초기화(랜덤)
  speedY = 0;						// 볼 y축 속도 초기화
}

function draw() {
  background(0);
  fill(255);
  ellipse(ballX, ballY, ballSize, ballSize);		// 볼 표시
  
  playerX = constrain(mouseX, w/2, width-w/2);	// 플레이어의 좌우 이동 제한
  playerY = constrain(mouseY, h/2, height-h/2);	// 플레이어의 상하 이동 제한

  fill(0, 255, 0);
  rectMode(CENTER);			// 사각형의 정가운데를 기준으로 폭과 높이 설정
  rect(playerX, playerY, w, h);		// 사각형 플레이어(라켓) 표시
  
  ballX += speedX;    		// x축 속도 누적 → x축 거리 이동 
  speedY += gravity;    		// y축 속도에 중력 누적 → 낙하 시 가속
  ballY += speedY;   		 	// y축 속도 누적 → y축 거리 이동

  if (ballX < ballSize/2 || ballX > (width - ballSize/2)) {	// 볼이 좌우 경계에 충돌 시
    speedX *= -1;			// 볼 속도의 방향 반전, 즉 반사
  }
  if (ballX > (playerX - w/2) && ballX < (playerX + w/2) && ballY > (playerY - (ballSize + h) / 2)) {			// 플레이어로 볼을 치는 경우
    ballY = playerY - (ballSize + h) / 2;	// ballY 위치는 라켓 상단(→[참고16.1] 참조)
    speedY *= -1;    			// 반사 적용
    speedY *= friction;			// (감쇄)마찰율 적용
    if (speedY > -1) speedY = -10;	// 볼이 거의 반사되지 않고 정지할 경우 강제 재점프
  }
  
  if (ballY > height) {			// 볼이 캔버스 바닥을 벗어난 경우
    fill(255);
    textSize(30);
    textAlign(CENTER, CENTER);	// 문자열 가운데 정렬
    text('GAME OVER', width/2, height/2);
    noLoop();				// draw() 함수의 반복 실행 일시 정지
  }
}
