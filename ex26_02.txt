// [본문 코드] 원형 객체 좌우 반사
// 객체 두 개 생성 (완성)

let c1, c2;

function setup() {
  createCanvas(300, 300);
  c1 = new Circle(0, 100);
  c2 = new Circle(50, 200);
}

function draw() {
  background(0, 10);	// 페이딩 효과
  c1.display();		// 메소드 호출
  c1.move();		// 메소드 호출
  c2.display();		// 메소드 호출
  c2.move();		// 메소드 호출
}

// [클래스 코드]

class Circle {
  constructor(x, y) {	// 생성자
    this.c = color(255);
    this.x = x;		// 매개변수로 초기화
    this.y = y;		// 매개변수로 초기화
    this.d = 30;
    this.speed = 1;
  }
  display() {		// 메소드
    noStroke();
    fill(this.c);
    ellipse(this.x, this.y, this.d, this.d);
  }
  move() {		// 메소드
    this.x += this.speed;
    if (this.x < 0 || this.x > width) {
      this.speed *= -1;
    }
  }
}
